export const dynamic = "force-dynamic";

// Strict allowlist of supported worker endpoint paths and their permitted HTTP methods
const ALLOWED_ENDPOINTS: Record<string, string[]> = {
  health: ["GET"],
  status: ["GET"],
  enqueue: ["POST"],
  enqueue_hf: ["POST"],
  shutdown: ["POST"],
  pause: ["POST"],
  resume: ["POST"],
  cancel: ["POST"],
};

function getCleanEndpoint(rawSegments: string[]): string {
  let segments = [...rawSegments];
  if (segments[0] === "api" && segments[1] === "v1") {
    segments = segments.slice(2);
  } else if (segments[0] === "api") {
    segments = segments.slice(1);
  }
  return segments.join("/").toLowerCase();
}

export async function GET(
  req: NextRequest,
  context: { params: Promise<{ path: string[] }> }
) {
  // 1. Enforce Server-Side Authentication
  const auth = await verifyServerAuth(req);
  if (!auth.authorized) {
    return auth.response;
  }

  const { path } = await context.params;
  const endpoint = getCleanEndpoint(path);

  // 2. Route Allowlist Validation
  if (!ALLOWED_ENDPOINTS[endpoint]) {
    return NextResponse.json(
      { error: `Endpoint '/${endpoint}' is not recognized or supported by worker gateway.` },
      { status: 404 }
    );
  }

  if (!ALLOWED_ENDPOINTS[endpoint].includes("GET")) {
    return NextResponse.json(
      { error: `Method GET is not permitted for '/${endpoint}'.` },
      { status: 405, headers: { Allow: ALLOWED_ENDPOINTS[endpoint].join(", ") } }
    );
  }

  // 3. Resolve Target Worker URL
  const env = getServerEnv();
  const backendBase = env.BACKEND_URL?.replace(/\/+$/, "");

  if (!backendBase) {
    return NextResponse.json(
      { error: "Configuration Error: BACKEND_URL is not set on the server." },
      { status: 503 }
    );
  }

  const search = req.nextUrl.search || "";
  const targetUrl = `${backendBase}/api/v1/${endpoint}${search}`;

  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), 8000);

  try {
    const res = await fetch(targetUrl, {
      method: "GET",
      headers: {
        "ngrok-skip-browser-warning": "69420",
        Accept: "application/json",
      },
      signal: controller.signal,
      cache: "no-store",
    });

    if (!res.ok) {
      return NextResponse.json(
        {
          status: "unavailable",
          message: `Worker returned HTTP ${res.status}`,
          upstreamStatus: res.status,
        },
        { status: 502 }
      );
    }

    const data = await res.json().catch(() => null);
    if (!data) {
      return NextResponse.json(
        { status: "degraded", message: "Worker returned invalid non-JSON payload" },
        { status: 502 }
      );
    }

    return NextResponse.json(data);
  } catch (error: any) {
    const isTimeout = error.name === "AbortError";
    return NextResponse.json(
      {
        status: "offline",
        message: isTimeout ? "Connection to remote worker timed out" : "Unable to reach remote worker tunnel",
        isTimeout,
      },
      { status: 504 }
    );
  } finally {
    clearTimeout(timeoutId);
  }
}

export async function POST(
  req: NextRequest,
  context: { params: Promise<{ path: string[] }> }
) {
  // 1. Enforce Server-Side Authentication
  const auth = await verifyServerAuth(req, { requiredRole: "operator" });
  if (!auth.authorized) {
    return auth.response;
  }

  const { path } = await context.params;
  const endpoint = getCleanEndpoint(path);

  // 2. Route Allowlist Validation
  if (!ALLOWED_ENDPOINTS[endpoint]) {
    return NextResponse.json(
      { error: `Endpoint '/${endpoint}' is not recognized or supported by worker gateway.` },
      { status: 404 }
    );
  }

  if (!ALLOWED_ENDPOINTS[endpoint].includes("POST")) {
    return NextResponse.json(
      { error: `Method POST is not permitted for '/${endpoint}'.` },
      { status: 405, headers: { Allow: ALLOWED_ENDPOINTS[endpoint].join(", ") } }
    );
  }

  // 3. Rate Limit Enqueue and Action Commands
  const rateLimitKey = `worker:${auth.user.uid}:${endpoint}`;
  const rateLimit = checkRateLimit(rateLimitKey, {
    windowMs: 60 * 1000,
    maxRequests: endpoint.startsWith("enqueue") ? 20 : 10,
  });

  if (!rateLimit.allowed) {
    return NextResponse.json(
      {
        error: "Too Many Requests",
        message: `Rate limit exceeded for '/${endpoint}'. Please slow down.`,
        retryAfterMs: rateLimit.resetInMs,
      },
      { status: 429 }
    );
  }

  // 4. Resolve Target Worker URL
  const env = getServerEnv();
  const backendBase = env.BACKEND_URL?.replace(/\/+$/, "");

  if (!backendBase) {
    return NextResponse.json(
      { error: "Configuration Error: BACKEND_URL is not set on the server." },
      { status: 503 }
    );
  }

  const search = req.nextUrl.search || "";
  const targetUrl = `${backendBase}/api/v1/${endpoint}${search}`;

  const contentType = req.headers.get("content-type") || "";
  const headers: Record<string, string> = {
    "ngrok-skip-browser-warning": "69420",
    Accept: "application/json",
  };

  let bodyPayload: any;

  try {
    if (
      contentType.includes("multipart/form-data") ||
      contentType.includes("application/x-www-form-urlencoded")
    ) {
      bodyPayload = await req.formData();
    } else if (contentType.includes("application/json")) {
      const rawText = await req.text();
      // Enforce max body size 2MB
      if (rawText.length > 2 * 1024 * 1024) {
        return NextResponse.json({ error: "Payload Too Large (Max 2MB)" }, { status: 413 });
      }
      bodyPayload = rawText;
      headers["Content-Type"] = "application/json";
    } else {
      bodyPayload = await req.text();
    }
  } catch {
    return NextResponse.json({ error: "Invalid request payload format" }, { status: 400 });
  }

  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), 15000);

  try {
    const res = await fetch(targetUrl, {
      method: "POST",
      headers,
      body: bodyPayload,
      signal: controller.signal,
      cache: "no-store",
    });

    const data = await res.json().catch(() => ({}));
    return NextResponse.json(data, { status: res.status });
  } catch (error: any) {
    const isTimeout = error.name === "AbortError";
    return NextResponse.json(
      {
        status: "error",
        message: isTimeout ? "Worker request timed out" : "Failed to communicate with remote Codespace worker",
        isTimeout,
      },
      { status: 504 }
    );
  } finally {
    clearTimeout(timeoutId);
  }
}

import { NextRequest, NextResponse } from "next/server";
import { triggerCodespaceWorkflow } from "@/lib/github";
import { verifyServerAuth } from "@/lib/serverAuth";
import { checkRateLimit } from "@/lib/rateLimit";

export const dynamic = "force-dynamic";

export async function POST(req: NextRequest) {
  // 1. Enforce Server-Side Authentication & Authorization
  const auth = await verifyServerAuth(req, { requiredRole: "operator" });
  if (!auth.authorized) {
    return auth.response;
  }

  // 2. Enforce Rate Limiting & Cooldown (Max 2 wake attempts per 60 seconds)
  const clientIdentifier = `wake:${auth.user.uid || req.headers.get("x-forwarded-for") || "unknown"}`;
  const rateLimit = checkRateLimit(clientIdentifier, {
    windowMs: 60 * 1000,
    maxRequests: 2,
  });

  if (!rateLimit.allowed) {
    return NextResponse.json(
      {
        error: "Too Many Requests",
        message: "Codespace wake is on cooldown. Please wait before triggering another wake sequence.",
        retryAfterMs: rateLimit.resetInMs,
      },
      {
        status: 429,
        headers: {
          "Retry-After": String(Math.ceil(rateLimit.resetInMs / 1000)),
        },
      }
    );
  }

  // 3. Dispatch GitHub Actions Workflow
  try {
    const result = await triggerCodespaceWorkflow();

    if (!result.success) {
      return NextResponse.json(
        { error: result.message || "Failed to trigger wake workflow" },
        { status: 502 }
      );
    }

    return NextResponse.json({
      status: "dispatched",
      message: result.message,
      actor: auth.user.email,
      timestamp: new Date().toISOString(),
    });
  } catch (err: any) {
    return NextResponse.json(
      { error: "Failed to initiate Codespace wake sequence" },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import { google } from "googleapis";
import { Readable } from "stream";
import { getServerEnv } from "@/lib/env";
import { verifyServerAuth } from "@/lib/serverAuth";

export const dynamic = "force-dynamic";

function cleanPrivateKey(key?: string): string {
  if (!key) return "";
  let cleaned = key.trim();
  if (
    (cleaned.startsWith('"') && cleaned.endsWith('"')) ||
    (cleaned.startsWith("'") && cleaned.endsWith("'"))
  ) {
    cleaned = cleaned.slice(1, -1);
  }
  return cleaned.replace(/\\n/g, "\n").replace(/\r/g, "").trim();
}

function getDriveClient() {
  const env = getServerEnv();
  const clientEmail = env.GDRIVE_CLIENT_EMAIL;
  const rawKey = env.GDRIVE_PRIVATE_KEY;

  if (!clientEmail || !rawKey) {
    throw new Error("Google Drive credentials not configured.");
  }

  const auth = new google.auth.JWT({
    email: clientEmail,
    key: cleanPrivateKey(rawKey),
    scopes: ["https://www.googleapis.com/auth/drive.readonly"],
  });

  return google.drive({ version: "v3", auth });
}

export async function GET(req: NextRequest) {
  // 1. Authenticate request (supports Authorization header or ?token query param)
  const auth = await verifyServerAuth(req);
  if (!auth.authorized) {
    return auth.response;
  }

  try {
    const { searchParams } = new URL(req.url);
    const fileId = searchParams.get("fileId");

    if (!fileId || typeof fileId !== "string" || fileId.length < 5) {
      return NextResponse.json({ error: "Missing or invalid fileId parameter" }, { status: 400 });
    }

    const env = getServerEnv();
    const rootFolderId = env.GDRIVE_FOLDER_ID;
    const drive = getDriveClient();

    // 2. Fetch metadata to inspect parents and MIME type
    const metadata = await drive.files.get({
      fileId,
      fields: "id, name, mimeType, size, parents, trashed",
      supportsAllDrives: true,
    });

    if (!metadata.data || metadata.data.trashed) {
      return NextResponse.json({ error: "File not found or was moved to trash" }, { status: 404 });
    }

    // 3. Prevent unauthorized traversal outside root folder
    if (rootFolderId) {
      const parents = metadata.data.parents || [];
      // Fast check: direct parent or ancestor
      if (parents.length > 0 && !parents.includes(rootFolderId)) {
        // Check ancestor chain
        let isChild = false;
        let queue = [...parents];
        const visited = new Set<string>();

        while (queue.length > 0 && !isChild) {
          const pid = queue.shift()!;
          if (pid === rootFolderId) {
            isChild = true;
            break;
          }
          if (visited.has(pid)) continue;
          visited.add(pid);

          try {
            const pMeta = await drive.files.get({
              fileId: pid,
              fields: "parents",
              supportsAllDrives: true,
            });
            if (pMeta.data.parents) queue.push(...pMeta.data.parents);
          } catch {
            // Stop traversal if ancestor cannot be read
          }
        }

        if (!isChild) {
          return NextResponse.json(
            { error: "Forbidden: File is outside the authorized storage tree" },
            { status: 403 }
          );
        }
      }
    }

    // 4. Download media stream
    const driveRes = await drive.files.get(
      { fileId, alt: "media", supportsAllDrives: true },
      { responseType: "stream" }
    );

    const headers = new Headers();
    const fileName = metadata.data.name || "download";
    headers.set(
      "Content-Disposition",
      `attachment; filename="${encodeURIComponent(fileName)}"`
    );
    headers.set(
      "Content-Type",
      metadata.data.mimeType || "application/octet-stream"
    );
    headers.set("Cache-Control", "private, no-store, max-age=0");

    if (metadata.data.size) {
      headers.set("Content-Length", metadata.data.size);
    }

    const webStream = Readable.toWeb(driveRes.data as unknown as Readable);
    return new Response(webStream as BodyInit, { headers });
  } catch (error: any) {
    console.error("[Drive Download Error]", error);
    return NextResponse.json(
      { error: error.message || "Failed to download file from Google Drive" },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import { google } from "googleapis";
import { getServerEnv } from "@/lib/env";
import { verifyServerAuth } from "@/lib/serverAuth";

export const dynamic = "force-dynamic";

function cleanPrivateKey(key?: string): string {
  if (!key) return "";
  let cleaned = key.trim();
  if (
    (cleaned.startsWith('"') && cleaned.endsWith('"')) ||
    (cleaned.startsWith("'") && cleaned.endsWith("'"))
  ) {
    cleaned = cleaned.slice(1, -1);
  }
  return cleaned.replace(/\\n/g, "\n").replace(/\r/g, "").trim();
}

function getDriveClient() {
  const env = getServerEnv();
  const clientEmail = env.GDRIVE_CLIENT_EMAIL;
  const privateKey = cleanPrivateKey(env.GDRIVE_PRIVATE_KEY);

  if (!clientEmail || !privateKey) {
    throw new Error("Missing Google Drive credentials in environment (GDRIVE_CLIENT_EMAIL / GDRIVE_PRIVATE_KEY)");
  }

  const auth = new google.auth.JWT({
    email: clientEmail,
    key: privateKey,
    scopes: ["https://www.googleapis.com/auth/drive.readonly"],
  });

  return google.drive({ version: "v3", auth });
}

// In-memory cache for Drive file listing to avoid quota exhaustion from rapid polling
interface DriveCache {
  data: any;
  timestamp: number;
}
const cacheStore = new Map<string, DriveCache>();
const CACHE_TTL_MS = 15000; // 15 seconds cache

export async function GET(req: NextRequest) {
  // 1. Authenticate caller
  const auth = await verifyServerAuth(req);
  if (!auth.authorized) {
    return auth.response;
  }

  try {
    const env = getServerEnv();
    const rootFolderId = env.GDRIVE_FOLDER_ID;

    if (!rootFolderId) {
      return NextResponse.json(
        { error: "Configuration Error: GDRIVE_FOLDER_ID is not configured." },
        { status: 500 }
      );
    }

    const { searchParams } = new URL(req.url);
    const requestedFolderId = searchParams.get("folderId") || rootFolderId;
    const search = searchParams.get("search");

    const cacheKey = `drive:${requestedFolderId}:${search || ""}`;
    const cached = cacheStore.get(cacheKey);
    if (cached && Date.now() - cached.timestamp < CACHE_TTL_MS) {
      return NextResponse.json(cached.data);
    }

    const drive = getDriveClient();

    // 2. Fetch files with full pagination
    let allFiles: any[] = [];
    let pageToken: string | undefined = undefined;
    let pageCount = 0;
    const maxPages = 5; // Safety cap at 5,000 items

    do {
      const listResponse: any = await drive.files.list({
        q: "trashed = false",
        fields: "nextPageToken, files(id, name, mimeType, size, parents, modifiedTime, webViewLink, iconLink)",
        pageSize: 1000,
        pageToken,
        supportsAllDrives: true,
        includeItemsFromAllDrives: true,
      });

      if (listResponse.data.files) {
        allFiles.push(...listResponse.data.files);
      }
      pageToken = listResponse.data.nextPageToken || undefined;
      pageCount++;
    } while (pageToken && pageCount < maxPages);

    // 3. Build ancestor and parent map to enforce root folder boundary
    const parentMap = new Map<string, string[]>();
    for (const item of allFiles) {
      if (item.id && item.parents) {
        parentMap.set(item.id, item.parents);
      }
    }

    // Helper to verify if an item belongs under rootFolderId
    const isDescendantOfRoot = (itemId: string): boolean => {
      if (itemId === rootFolderId) return true;
      const visited = new Set<string>();
      let queue = parentMap.get(itemId) ? [...(parentMap.get(itemId) || [])] : [];

      while (queue.length > 0) {
        const parentId = queue.shift()!;
        if (parentId === rootFolderId) return true;
        if (visited.has(parentId)) continue;
        visited.add(parentId);
        const grandparents = parentMap.get(parentId);
        if (grandparents) queue.push(...grandparents);
      }
      return false;
    };

    // Verify requested folder belongs to authorized root tree
    if (requestedFolderId !== rootFolderId && !isDescendantOfRoot(requestedFolderId)) {
      return NextResponse.json(
        { error: "Forbidden: Requested folder is outside the authorized storage tree." },
        { status: 403 }
      );
    }

    // 4. Calculate real sizes and rolling time window totals
    const now = Date.now();
    const twentyFourHoursAgo = now - 24 * 60 * 60 * 1000;
    const startOfTodayMs = new Date().setUTCHours(0, 0, 0, 0);

    let bytes24h = 0;
    let bytesToday = 0;
    let totalStorageBytes = 0;
    const folderSizeMap = new Map<string, number>();

    for (const item of allFiles) {
      // Only account for files within the authorized root scope
      if (!isDescendantOfRoot(item.id || "")) continue;

      const isFolder = item.mimeType === "application/vnd.google-apps.folder";
      if (!isFolder && item.size) {
        const bytes = parseInt(item.size, 10) || 0;
        totalStorageBytes += bytes;

        const modTime = item.modifiedTime ? new Date(item.modifiedTime).getTime() : 0;
        if (modTime >= twentyFourHoursAgo) bytes24h += bytes;
        if (modTime >= startOfTodayMs) bytesToday += bytes;

        // Bubble up bytes to ancestor folders
        let queue = item.parents ? [...item.parents] : [];
        const visited = new Set<string>();

        while (queue.length > 0) {
          const parentId = queue.shift()!;
          if (visited.has(parentId)) continue;
          visited.add(parentId);

          folderSizeMap.set(parentId, (folderSizeMap.get(parentId) || 0) + bytes);
          const grandparents = parentMap.get(parentId);
          if (grandparents) queue.push(...grandparents);
        }
      }
    }

    // 5. Filter items for display
    let displayItems: any[] = [];
    if (search && search.trim()) {
      const term = search.trim().toLowerCase();
      displayItems = allFiles.filter(
        (item) => isDescendantOfRoot(item.id || "") && (item.name || "").toLowerCase().includes(term)
      );
    } else {
      displayItems = allFiles.filter(
        (item) => item.parents && item.parents.includes(requestedFolderId)
      );
    }

    const files = displayItems.map((file) => {
      const isFolder = file.mimeType === "application/vnd.google-apps.folder";
      const sizeBytes = isFolder
        ? folderSizeMap.get(file.id || "") || 0
        : file.size
        ? parseInt(file.size, 10)
        : 0;

      return {
        id: file.id || "",
        name: file.name || "Untitled",
        mimeType: file.mimeType || "",
        isFolder,
        sizeBytes,
        modifiedTime: file.modifiedTime || new Date().toISOString(),
        webViewLink: file.webViewLink || "",
        iconLink: file.iconLink || "",
      };
    });

    const folderStorage = folderSizeMap.get(requestedFolderId);
    const directChildrenCount = displayItems.length;

    const responsePayload = {
      success: true,
      folderId: requestedFolderId,
      totalStorageBytes: typeof folderStorage === "number" ? folderStorage : totalStorageBytes,
      bytes24h,
      bytesToday,
      totalCompletedCount: directChildrenCount,
      files,
    };

    cacheStore.set(cacheKey, { data: responsePayload, timestamp: Date.now() });
    return NextResponse.json(responsePayload);
  } catch (error: any) {
    console.error("[Drive API Error]", error);
    return NextResponse.json(
      { error: "Failed to query Google Drive API" },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import {
  doc,
  getDoc,
  setDoc,
  updateDoc,
  increment,
  serverTimestamp,
} from "firebase/firestore";
import { db } from "@/lib/firebase";
import { verifyServerAuth } from "@/lib/serverAuth";
import { z } from "zod";

export const dynamic = "force-dynamic";

const taskCompletionSchema = z.object({
  taskId: z.string().min(1, "taskId is required for idempotent completion recording"),
  bytes: z.number().int().nonnegative("bytes must be a non-negative integer"),
  name: z.string().optional(),
  source: z.string().optional(),
});

export async function GET(req: NextRequest) {
  // 1. Authenticate request
  const auth = await verifyServerAuth(req);
  if (!auth.authorized) {
    return auth.response;
  }

  try {
    const docRef = doc(db, "analytics", "dashboard");
    const snap = await getDoc(docRef);

    if (!snap.exists()) {
      return NextResponse.json({
        completed_count: 0,
        total_bytes_extracted: 0,
        hourly_transfers: {},
      });
    }

    return NextResponse.json(snap.data());
  } catch (error: any) {
    return NextResponse.json({ error: error.message || "Failed to load metrics" }, { status: 500 });
  }
}

export async function POST(req: NextRequest) {
  // 1. Authenticate and authorize operator
  const auth = await verifyServerAuth(req, { requiredRole: "operator" });
  if (!auth.authorized) {
    return auth.response;
  }

  let body: any;
  try {
    body = await req.json();
  } catch {
    return NextResponse.json({ error: "Invalid JSON body" }, { status: 400 });
  }

  // 2. Validate input schema with Zod
  const parseResult = taskCompletionSchema.safeParse(body);
  if (!parseResult.success) {
    return NextResponse.json(
      { error: "Validation Error", details: parseResult.error.format() },
      { status: 400 }
    );
  }

  const { taskId, bytes, name, source } = parseResult.data;

  try {
    // 3. Idempotency Check: Verify task hasn't already been committed
    const taskLogRef = doc(db, "completed_tasks", taskId);
    const existingTask = await getDoc(taskLogRef);

    if (existingTask.exists()) {
      return NextResponse.json({
        success: true,
        message: "Task completion was already recorded (idempotent skip).",
        duplicate: true,
      });
    }

    // 4. Record task completion document atomically
    await setDoc(taskLogRef, {
      taskId,
      bytes,
      name: name || "Unnamed Task",
      source: source || "Cloud Stream",
      recordedBy: auth.user.email,
      completedAt: serverTimestamp(),
    });

    // 5. Update aggregate dashboard statistics
    const dashboardRef = doc(db, "analytics", "dashboard");
    const currentHour = new Date().getUTCHours().toString().padStart(2, "0");

    try {
      await updateDoc(dashboardRef, {
        completed_count: increment(1),
        total_bytes_extracted: increment(bytes),
        [`hourly_transfers.${currentHour}`]: increment(bytes),
        last_updated: serverTimestamp(),
      });
    } catch (err: any) {
      if (err.code === "not-found") {
        await setDoc(dashboardRef, {
          completed_count: 1,
          total_bytes_extracted: bytes,
          hourly_transfers: { [currentHour]: bytes },
          last_updated: serverTimestamp(),
        });
      } else {
        throw err;
      }
    }

    return NextResponse.json({ success: true, taskId, bytesRecorded: bytes });
  } catch (error: any) {
    return NextResponse.json(
      { error: error.message || "Failed to commit metrics record" },
      { status: 500 }
    );
  }
}

function getFileIcon(name: string, mimeType: string) {
  const lower = name.toLowerCase();

  if (
    lower.endsWith(".safetensors") ||
    lower.endsWith(".bin") ||
    lower.endsWith(".pt") ||
    lower.endsWith(".pth") ||
    lower.endsWith(".onnx") ||
    lower.endsWith(".gguf")

    return <Binary className="w-4 h-4 text-purple-400 shrink-0" />;
  }

  if (
    lower.endsWith(".parquet") ||
    lower.endsWith(".arrow") ||
    lower.endsWith(".csv") ||
    lower.endsWith(".json") ||
    lower.endsWith(".jsonl")
  ) {
    return <Database className="w-4 h-4 text-emerald-400 shrink-0" />;
  }

  if (
    lower.endsWith(".mkv") ||
    lower.endsWith(".mp4") ||
    lower.endsWith(".avi") ||
    lower.endsWith(".mov") ||
    mimeType.startsWith("video/")
  ) {
    return <Film className="w-4 h-4 text-amber-400 shrink-0" />;
  }

  if (
    lower.endsWith(".zip") ||
    lower.endsWith(".tar") ||
    lower.endsWith(".gz") ||
    lower.endsWith(".7z") ||
    lower.endsWith(".rar")
  ) {
    return <FileArchive className="w-4 h-4 text-rose-400 shrink-0" />;
  }

  if (
    lower.endsWith(".py") ||
    lower.endsWith(".ts") ||
    lower.endsWith(".js") ||
    lower.endsWith(".sh") ||
    lower.endsWith(".yaml") ||
    lower.endsWith(".yml")
  ) {
    return <FileCode className="w-4 h-4 text-cyan-400 shrink-0" />;

      const url = `/api/drive/download?fileId=${encodeURIComponent(item.id)}&name=${encodeURIComponent(
        item.name
      )}${token ? `&token=${encodeURIComponent(token)}` : ""}`;
      const a = document.createElement("a");
      a.href = url;
      a.download = item.name;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
    } catch (err) {
      console.error("Download failed:", err);
    }
  };

  const updateTreeWithChildren = (
    list: DriveItem[],
    targetId: string,
    children: DriveItem[],
    isLoading: boolean
  ): DriveItem[] => {
    return list.map((node) => {
      if (node.id === targetId) {
        return { ...node, children, isLoadingChildren: isLoading };
      }
      if (node.children && node.children.length > 0) {
        return {
          ...node,
          children: updateTreeWithChildren(node.children, targetId, children, isLoading),
        };
      }
      return node;
    });
  };

  const setTreeLoading = (
    list: DriveItem[],
    targetId: string,
    isLoading: boolean
  ): DriveItem[] => {
    return list.map((node) => {
      if (node.id === targetId) {
        return { ...node, isLoadingChildren: isLoading };
      }
      if (node.children && node.children.length > 0) {
        return {
          ...node,
          children: setTreeLoading(node.children, targetId, isLoading),
        };
      }
      return node;
    });
  };

  const handleToggleFolder = async (item: DriveItem) => {
    const isCurrentlyExpanded = !!expandedFolders[item.id];

    setExpandedFolders((prev) => ({
      ...prev,
      [item.id]: !isCurrentlyExpanded,
    }));

    if (!isCurrentlyExpanded && (!item.children || item.children.length === 0)) {
      const updateList = (fn: (prev: DriveItem[]) => DriveItem[]) => {
        setRootFiles((prev) => fn(prev));
        setSearchResults((prev) => (prev ? fn(prev) : null));
      };

      updateList((prev) => setTreeLoading(prev, item.id, true));

      try {
        const headers = await getAuthHeaders();
        const res = await fetch(`/api/drive?folderId=${encodeURIComponent(item.id)}`, {
          cache: "no-store",
          headers,
        });
        const subData = await res.json();
        const parsedChildren = parseItems(subData.files || []);

        updateList((prev) =>
          updateTreeWithChildren(prev, item.id, parsedChildren, false)
        );
      } catch {
        updateList((prev) => setTreeLoading(prev, item.id, false));
      }
    }
  };

  const usedPercentage = Math.min(
    parseFloat(((totalDriveBytes / TOTAL_CAPACITY_BYTES) * 100).toFixed(1)),
    100
  );
  const remainingBytes = Math.max(TOTAL_CAPACITY_BYTES - totalDriveBytes, 0);

  const isSearchActive = searchTerm.trim().length > 0 && searchResults !== null;
  const displayedFiles = isSearchActive ? searchResults : rootFiles;

      {/* Header */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 sm:gap-4">
        <div>
          <h1 className="text-xl sm:text-2xl font-bold tracking-tight text-white flex items-center gap-2 sm:gap-2.5">
            <HardDrive className="w-5 h-5 sm:w-6 sm:h-6 text-blue-500" />
            <span>Google Drive Storage</span>
          </h1>
          <p className="text-[11px] sm:text-xs md:text-sm text-slate-400 mt-0.5 sm:mt-1">
            Browse and manage files streamed directly through Rclone.
          </p>

            variant="secondary"
            size="sm"
            onClick={handleRefresh}
            disabled={isRefreshing}
            className="flex items-center gap-1.5 h-8 text-xs"
          >
            <RefreshCw
              className={`w-3.5 h-3.5 text-slate-400 ${

            size="sm"

            <ExternalLink className="w-3.5 h-3.5" />
          </Button>
        </div>
      </div>

      {/* Error Alert */}

import React, { useMemo } from "react";

interface AnalyticsGridProps {
  speedHistory?: SpeedDataPoint[];
  currentSpeed?: string;
  bytes24h?: number;
  hourlyData?: Record<string, number>;
}

export const AnalyticsGrid: React.FC<AnalyticsGridProps> = ({
  speedHistory = [],
  currentSpeed = "0.00 MB/s",
  bytes24h = 0,
  hourlyData,
}) => {
  const chartPoints = useMemo(() => {
    const width = 300;
    const height = 90;
    const padding = 5;

    if (!speedHistory || speedHistory.length === 0) {
      return {
        path: `M 0 ${height - padding} L ${width} ${height - padding}`,
        area: `M 0 ${height - padding} L ${width} ${height - padding} L ${width} ${height} L 0 ${height} Z`,
      };
    }

    const maxSpeed = Math.max(
      ...speedHistory.map((d) => Number(d.speed) || 0),
      50
    );

    const safePoints =
      speedHistory.length === 1
        ? [speedHistory[0], speedHistory[0]]
        : speedHistory;

    const stepX = width / Math.max(safePoints.length - 1, 1);

    const coords = safePoints.map((d, index) => {
      const speedVal = Number(d.speed) || 0;
      const x = index * stepX;
      const normalizedY = (speedVal / maxSpeed) * (height - padding * 2);
      const y = height - padding - normalizedY;
      return { x, y };

"use client";

import React, { useState, useEffect } from "react";
import {
  ArrowRight,
  Link2,
  Sparkles,
  Database,
  Globe,
  Loader2,
  AlertCircle,
} from "lucide-react";
import { Button } from "@/components/ui/button";
import { validateExtractionInput } from "@/lib/validation";

export type SourceBadge = "auto" | "hf" | "academic";
export type ExecutionMode = "cloud" | "local";

interface ExtractionCardProps {
  onExtract: (payload: {
    url: string;
    source: SourceBadge;
    mode: ExecutionMode;
  }) => Promise<void> | void;
  isSubmitting?: boolean;
}

export const ExtractionCard: React.FC<ExtractionCardProps> = ({
  onExtract,
  isSubmitting = false,
}) => {
  const [url, setUrl] = useState("");
  const [source, setSource] = useState<SourceBadge>("auto");
  const [mode, setMode] = useState<ExecutionMode>("cloud");
  const [validationError, setValidationError] = useState<string | null>(null);

  // Listen for QuickActions clipboard/preset injection
  useEffect(() => {
    const handleSetUrl = (e: Event) => {
      const customEvent = e as CustomEvent<string>;
      if (customEvent.detail) {
        const val = customEvent.detail.trim();
        setUrl(val);
        setValidationError(null);

        if (val.includes("huggingface.co") || val.includes("hf.co")) {
          setSource("hf");
        } else if (
          val.startsWith("magnet:?") ||
          val.includes("academictorrents") ||
          val.endsWith(".torrent")
        ) {
          setSource("academic");
        }
      }
    };

    window.addEventListener("runsync:set_url", handleSetUrl);
    return () => window.removeEventListener("runsync:set_url", handleSetUrl);
  }, []);

  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    setUrl(e.target.value);
    if (validationError) {
      setValidationError(null);
    }
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!url.trim() || isSubmitting) return;

    // Validate using strict schema
    const validation = validateExtractionInput(url, source);
    if (!validation.valid) {
      setValidationError(validation.error || "Invalid extraction input.");
      return;
    }

    setValidationError(null);

    await onExtract({
      url: validation.normalizedInput || url.trim(),
      source,
      mode,
    });

    setUrl("");
  };

  return (
    <div className="p-4 sm:p-5 md:p-6 rounded-xl sm:rounded-2xl bg-[#0D1424] border border-[#1A263D] shadow-xl space-y-3.5 sm:space-y-4">
      <div className="flex items-center justify-between">
        <span className="text-xs font-medium text-slate-400">New Extraction</span>
        {isSubmitting && (
          <div className="flex items-center gap-1.5 text-blue-400 text-[11px] sm:text-xs font-mono">
            <Loader2 className="w-3.5 h-3.5 animate-spin" />
            <span className="truncate">Dispatching to cloud...</span>
          </div>
        )}
      </div>

      <form onSubmit={handleSubmit} className="space-y-3">
        {/* Main URL Input Bar */}
        <div className="relative flex items-center">
          <Link2 className="w-4 h-4 text-slate-500 absolute left-3 shrink-0" />
          <input
            type="text"
            value={url}
            onChange={handleInputChange}
            disabled={isSubmitting}
            placeholder="Paste infohash, magnet, or Hugging Face repository link..."
            className="w-full bg-[#070B14] border border-[#1A263D] focus:border-blue-500 rounded-xl pl-9 pr-24 sm:pr-28 py-2.5 text-xs text-slate-200 placeholder-slate-500 focus:outline-none transition-all font-mono disabled:opacity-60"
          />
          <Button
            type="submit"
            size="sm"
            variant="primary"
            disabled={!url.trim() || isSubmitting}
            className="absolute right-1.5 h-7 sm:h-8 px-2.5 sm:px-3.5 text-xs flex items-center gap-1 disabled:opacity-50"
          >
            {isSubmitting ? (
              <>
                <Loader2 className="w-3.5 h-3.5 animate-spin" />
                <span className="hidden sm:inline">Sending</span>
              </>
            ) : (
              <>
                <span>Extract</span>
                <ArrowRight className="w-3.5 h-3.5" />
              </>
            )}
          </Button>
        </div>

        {/* Validation Error Message */}
        {validationError && (
          <div className="flex items-center gap-1.5 text-rose-400 text-xs px-1">
            <AlertCircle className="w-3.5 h-3.5 shrink-0" />
            <span>{validationError}</span>
          </div>
        )}

        {/* Source Badges & Execution Mode Switcher */}
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2.5 pt-0.5">
          {/* Scrollable Badges */}
          <div className="flex items-center gap-1.5 overflow-x-auto pb-1 sm:pb-0 scrollbar-none no-scrollbar">
            <button
              type="button"
              disabled={isSubmitting}
              onClick={() => {
                setSource("auto");
                setValidationError(null);
              }}
              className={`px-2.5 py-1 rounded-lg text-[11px] sm:text-xs font-medium flex items-center gap-1.5 whitespace-nowrap transition-all shrink-0 ${
                source === "auto"
                  ? "bg-blue-600/20 text-blue-400 border border-blue-500/40"
                  : "bg-[#070B14] text-slate-400 border border-[#1A263D] hover:text-slate-200"
              }`}
            >
              <Sparkles className="w-3 h-3" />
              <span>Auto (Detection)</span>
            </button>

            <button
              type="button"
              disabled={isSubmitting}
              onClick={() => {
                setSource("hf");
                setValidationError(null);
              }}
              className={`px-2.5 py-1 rounded-lg text-[11px] sm:text-xs font-medium flex items-center gap-1.5 whitespace-nowrap transition-all shrink-0 ${
                source === "hf"
                  ? "bg-amber-600/20 text-amber-400 border border-amber-500/40"
                  : "bg-[#070B14] text-slate-400 border border-[#1A263D] hover:text-slate-200"
              }`}
            >
              <span>🤗 Hugging Face</span>
            </button>

            <button
              type="button"
              disabled={isSubmitting}
              onClick={() => {
                setSource("academic");
                setValidationError(null);
              }}
              className={`px-2.5 py-1 rounded-lg text-[11px] sm:text-xs font-medium flex items-center gap-1.5 whitespace-nowrap transition-all shrink-0 ${
                source === "academic"
                  ? "bg-rose-600/20 text-rose-400 border border-rose-500/40"
                  : "bg-[#070B14] text-slate-400 border border-[#1A263D] hover:text-slate-200"
              }`}
            >
              <span>🧲 Academic Torrents</span>
            </button>
          </div>

          {/* Mode Switcher */}
          <div className="flex items-center self-start sm:self-auto bg-[#070B14] p-0.5 rounded-lg border border-[#1A263D] shrink-0">
            <button
              type="button"
              disabled={isSubmitting}
              onClick={() => setMode("cloud")}
              className={`px-2.5 py-1 rounded-md text-[11px] sm:text-xs font-medium flex items-center gap-1.5 transition-all ${
                mode === "cloud"
                  ? "bg-blue-600 text-white shadow-md shadow-blue-600/30"
                  : "text-slate-400 hover:text-slate-200"
              }`}
            >
              <Database className="w-3 h-3" />
              <span>Cloud Extract</span>
            </button>

            <button
              type="button"
              disabled={isSubmitting}
              onClick={() => setMode("local")}
              className={`px-2.5 py-1 rounded-md text-[11px] sm:text-xs font-medium flex items-center gap-1.5 transition-all ${
                mode === "local"
                  ? "bg-blue-600 text-white shadow-md shadow-blue-600/30"
                  : "text-slate-400 hover:text-slate-200"
              }`}
            >
              <Globe className="w-3 h-3" />
              <span>Local Fetch</span>
            </button>
          </div>
        </div>
      </form>
    </div>
  );
};

"use client";

import React, { useState, useEffect } from "react";
import {
  signInWithPopup,
  signOut,
  onAuthStateChanged,
  User,
} from "firebase/auth";
import { auth, googleProvider } from "@/lib/firebase";
import { ShieldAlert, LogIn, Loader2, Zap, AlertCircle } from "lucide-react";

export function AuthGate({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState<boolean>(true);
  const [accessDenied, setAccessDenied] = useState<string | null>(null);
  const [authError, setAuthError] = useState<string | null>(null);

  const allowedEmails = (process.env.NEXT_PUBLIC_ALLOWED_EMAILS || "")
    .split(",")
    .map((e) => e.trim().toLowerCase())
    .filter(Boolean);

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, (currentUser) => {
      if (currentUser && currentUser.email) {
        const email = currentUser.email.toLowerCase();
        if (allowedEmails.length === 0 || allowedEmails.includes(email)) {
          setUser(currentUser);
          setAccessDenied(null);
          setAuthError(null);
        } else {
          signOut(auth);
          setUser(null);
          setAccessDenied(currentUser.email);
        }
      } else {
        setUser(null);
      }
      setLoading(false);
    });

    return () => unsubscribe();
  }, []);

  const handleGoogleLogin = async () => {
    setLoading(true);
    setAccessDenied(null);
    setAuthError(null);
    try {
      await signInWithPopup(auth, googleProvider);
    } catch (err: any) {
      console.error("Login failed:", err);
      setLoading(false);
      setAuthError(err.message || "Google authentication was cancelled or interrupted.");
    }
  };

  if (loading) {
    return (
      <div className="h-screen w-screen bg-[#080C14] flex flex-col items-center justify-center text-slate-400 gap-3">
        <Loader2 className="w-8 h-8 text-blue-500 animate-spin" />
        <span className="text-xs font-mono">Verifying authorization...</span>
      </div>
    );
  }

  if (!user) {
    return (
      <div className="h-screen w-screen bg-[#080C14] flex flex-col items-center justify-center p-4">
        <div className="w-full max-w-sm p-6 sm:p-8 rounded-2xl bg-[#0D1424] border border-[#1A263D] shadow-2xl flex flex-col items-center text-center space-y-5">
          <div className="w-12 h-12 rounded-2xl bg-gradient-to-tr from-blue-600 to-indigo-500 flex items-center justify-center text-white shadow-lg shadow-blue-500/30">
            <Zap className="w-6 h-6 fill-current" />
          </div>

          <div>
            <h1 className="text-xl font-bold text-white tracking-tight">RunSync 2.0</h1>
            <p className="text-xs text-slate-400 mt-1">
              Private Cloud Extraction & Swarm Pipeline
            </p>
          </div>

          {accessDenied && (
            <div className="w-full p-3 rounded-xl bg-rose-500/10 border border-rose-500/30 text-rose-300 text-xs flex items-center gap-2.5 text-left">
              <ShieldAlert className="w-4 h-4 shrink-0 text-rose-400" />
              <span>
                Access Denied for <strong>{accessDenied}</strong>. Email is not authorized.
              </span>
            </div>
          )}

          {authError && (
            <div className="w-full p-3 rounded-xl bg-amber-500/10 border border-amber-500/30 text-amber-300 text-xs flex items-center gap-2.5 text-left">
              <AlertCircle className="w-4 h-4 shrink-0 text-amber-400" />
              <span>{authError}</span>
            </div>
          )}

          <button
            onClick={handleGoogleLogin}
            className="w-full py-2.5 px-4 rounded-xl bg-blue-600 hover:bg-blue-500 active:scale-[0.99] text-white text-xs font-semibold flex items-center justify-center gap-2 transition-all shadow-lg shadow-blue-600/30"
          >
            <LogIn className="w-4 h-4" />
            <span>Sign in with Google</span>
          </button>
        </div>
      </div>
    );
  }

  return <>{children}</>;
}

import React from "react";
import { cn } from "@/lib/utils";

export type SparklineType = "line" | "bars";
export type SparklineColor = "blue" | "emerald" | "purple" | "amber";

interface SparklineProps {
  type?: SparklineType;
  color?: SparklineColor;
  data?: number[];
  className?: string;
  width?: number;
  height?: number;
}

export const Sparkline: React.FC<SparklineProps> = ({
  type = "line",
  color = "blue",
  data = [12, 18, 14, 25, 20, 28, 22, 34, 30],
  className,
  width = 96,
  height = 36,
}) => {
  const colorMap = {
    blue: {
      stroke: "#38BDF8",
      fill: "#0284C7",
      bar: "fill-blue-500",
      glow: "drop-shadow(0 0 4px rgba(56,189,248,0.5))",
    },
    emerald: {
      stroke: "#34D399",
      fill: "#059669",
      bar: "fill-emerald-500",
      glow: "drop-shadow(0 0 4px rgba(52,211,153,0.5))",
    },
    purple: {
      stroke: "#C084FC",
      fill: "#7E22CE",
      bar: "fill-purple-500",
      glow: "drop-shadow(0 0 4px rgba(192,132,252,0.5))",
    },
    amber: {
      stroke: "#FBBF24",
      fill: "#D97706",
      bar: "fill-amber-500",
      glow: "drop-shadow(0 0 4px rgba(251,191,36,0.5))",
    },
  };

  const selectedColor = colorMap[color];
  const safeData = data.length > 0 ? data : [0];

  // Renders vertical micro-bars (Completed & Extracted KPIs)
  if (type === "bars") {
    const maxVal = Math.max(...safeData, 1);
    const barWidth = 4;
    const gap = 4;
    const totalBars = safeData.length;
    const svgWidth = totalBars * (barWidth + gap);

    return (
      <svg
        width={svgWidth}
        height={height}
        viewBox={`0 0 ${svgWidth} ${height}`}
        className={cn("overflow-visible", className)}
      >
        {safeData.map((val, index) => {
          const barHeight = Math.max((val / maxVal) * (height - 6), 3);
          const x = index * (barWidth + gap);
          const y = height - barHeight;
          const opacity = 0.4 + (index / totalBars) * 0.6;

          return (
            <rect
              key={index}
              x={x}
              y={y}
              width={barWidth}
              height={barHeight}
              rx={2}
              className={selectedColor.bar}
              style={{ opacity }}
            />
          );
        })}
      </svg>
    );
  }

  // Renders smooth SVG wave curves (Active Transfers & Speed KPIs)
  const min = Math.min(...safeData);
  const max = Math.max(...safeData);
  const range = max - min || 1;
  const padding = 4;
  const usableHeight = height - padding * 2;

  // Safe point normalization (prevents divide-by-zero on single-item data)
  const safeDivisor = Math.max(safeData.length - 1, 1);

  const points = safeData.map((val, i) => {
    const x = (i / safeDivisor) * (width - 8) + 4;
    const y = height - padding - ((val - min) / range) * usableHeight;
    return { x, y };
  });

  let pathD = `M ${points[0].x} ${points[0].y}`;
  if (points.length === 1) {
    pathD += ` L ${width - 4} ${points[0].y}`;
  } else {
    for (let i = 0; i < points.length - 1; i++) {
      const curr = points[i];
      const next = points[i + 1];
      const cpX = (curr.x + next.x) / 2;
      pathD += ` C ${cpX} ${curr.y}, ${cpX} ${next.y}, ${next.x} ${next.y}`;
    }
  }

  return (
    <svg
      width={width}
      height={height}
      viewBox={`0 0 ${width} ${height}`}
      className={cn("overflow-visible", className)}
    >
      <path
        d={pathD}
        fill="none"
        stroke={selectedColor.stroke}
        strokeWidth={2.2}
        strokeLinecap="round"
        style={{ filter: selectedColor.glow }}
      />
    </svg>
  );
};

"use client";

import { useState, useEffect, useCallback, useRef } from "react";
import { checkBackendHealth, wakeCodespaceInstance } from "@/lib/api";

export type CodespaceStatusType = "online" | "booting" | "standby" | "degraded";

export function useCodespaceStatus(pollIntervalMs: number = 5000) {
  const [status, setStatus] = useState<CodespaceStatusType>("standby");
  const [cluster, setCluster] = useState<string>("Codespace • US-West");
  const [isBooting, setIsBooting] = useState<boolean>(false);
  const [engineState, setEngineState] = useState<string>("idle");
  const [lastCheckedTime, setLastCheckedTime] = useState<string>("");

  const isBootingRef = useRef(isBooting);
  isBootingRef.current = isBooting;

  const isCheckingRef = useRef(false);

  const checkStatus = useCallback(async (): Promise<boolean> => {
    if (isCheckingRef.current) return false;
    isCheckingRef.current = true;

    try {
      const health = await checkBackendHealth();
      setLastCheckedTime(new Date().toLocaleTimeString());

      if (health && health.status === "online") {
        setStatus("online");
        setIsBooting(false);
        if (health.cluster) setCluster(health.cluster);
        if (health.engine) setEngineState(health.engine);
        return true;
      } else if (health && health.status === "standby") {
        if (!isBootingRef.current) {
          setStatus("standby");
          setEngineState("idle");
        }
        return false;
      } else {
        // Outage or degraded response
        if (!isBootingRef.current) {
          setStatus((prev) => (prev === "online" ? "degraded" : "standby"));
        }
        return false;
      }
    } catch {
      if (!isBootingRef.current) {
        setStatus((prev) => (prev === "online" ? "degraded" : "standby"));
      }
      return false;
    } finally {
      isCheckingRef.current = false;
    }
  }, []);

  // Main unified polling loop using recursive timeout (no overlapping requests)
  useEffect(() => {
    let timeoutId: NodeJS.Timeout;
    let isMounted = true;

    const schedulePoll = () => {
      const isHidden = typeof document !== "undefined" && document.visibilityState === "hidden";
      const delay = isHidden ? 15000 : isBooting ? 2500 : pollIntervalMs;

      timeoutId = setTimeout(async () => {
        if (isMounted) {
          await checkStatus();
          if (isMounted) schedulePoll();
        }
      }, delay);
    };

    checkStatus().then(() => {
      if (isMounted) schedulePoll();
    });

    const handleVisibilityChange = () => {
      if (document.visibilityState === "visible") {
        checkStatus();
      }
    };

    if (typeof document !== "undefined") {
      document.addEventListener("visibilitychange", handleVisibilityChange);
    }

    return () => {
      isMounted = false;
      clearTimeout(timeoutId);
      if (typeof document !== "undefined") {
        document.removeEventListener("visibilitychange", handleVisibilityChange);
      }
    };
  }, [checkStatus, pollIntervalMs, isBooting]);

  /**
   * Dispatches GitHub Action to wake up Codespace
   */
  const wakeUpCodespace = useCallback(async (): Promise<boolean> => {
    setIsBooting(true);
    setStatus("booting");

    const dispatched = await wakeCodespaceInstance();
    if (!dispatched) {
      setIsBooting(false);
      setStatus("standby");
      return false;
    }

    // Immediately trigger status check
    await checkStatus();
    return true;
  }, [checkStatus]);

  return {
    status,
    isOnline: status === "online",
    isBooting,
    cluster,
    engineState,
    lastCheckedTime,
    checkStatus,
    wakeUpCodespace,
  };
}

function getLocalDateString(): string {
  const now = new Date();
  const year = now.getFullYear();
  const month = String(now.getMonth() + 1).padStart(2, "0");
  const day = String(now.getDate()).padStart(2, "0");

function parseBytesFromTaskName(name: string): number {
  if (!name) return 0;
  const match = name.match(/(\d+(?:\.\d+)?)\s*(gb|gib|mb|mib|kb|kib)/i);
  if (!match) return 0;

  const val = parseFloat(match[1]);
  const unit = match[2].toLowerCase();

  if (unit.startsWith("g")) return Math.round(val * 1024 * 1024 * 1024);
  if (unit.startsWith("m")) return Math.round(val * 1024 * 1024);
  if (unit.startsWith("k")) return Math.round(val * 1024);
  return 0;
}

export function parseSpeedStringToMbps(speedStr: string | undefined): number {
  if (!speedStr) return 0;
  const cleaned = speedStr.trim().toLowerCase();
  if (
    cleaned.includes("finish") ||
    cleaned.includes("idle") ||
    cleaned.includes("done") ||
    cleaned.includes("offline") ||
    cleaned.includes("calculating")
  ) {
    return 0;
  }

  const numericMatch = cleaned.match(/([\d.]+)/);
  if (!numericMatch) return 0;

  const rawVal = parseFloat(numericMatch[1]);
  if (isNaN(rawVal)) return 0;

  if (cleaned.includes("gb/s") || cleaned.includes("gib/s") || cleaned.includes("gbyte/s")) {
    return rawVal * 1024;
  }
  if (cleaned.includes("mb/s") || cleaned.includes("mib/s") || cleaned.includes("mbyte/s")) {
    return rawVal;
  }
  if (cleaned.includes("kb/s") || cleaned.includes("kib/s") || cleaned.includes("kbyte/s")) {
    return rawVal / 1024;
  }
  if (cleaned.includes("b/s") || cleaned.includes("byte/s")) {
    return rawVal / (1024 * 1024);
  }
  return rawVal;
}

            (sum: number, val: any) => sum + (Number(val) || 0),
            0
          ) as number;

          setFirestoreTotalBytes(totalBytes);
          setFirestoreBytes24h(total24h);
          setFirestoreBytesToday(todayBytes);
          setHourlyTransfers(hourly);

          setSnapshot((prev) => ({
            ...prev,
            hourlyTransfers: hourly,
            kpis: {
              ...prev.kpis,
              completed: {
                ...prev.kpis.completed,
                value: completedCount,
                trend: `+${completedWeek} this week`,
              },
              dataExtracted: {
                ...prev.kpis.dataExtracted,
                value: formatBytes(totalBytes),
                trend: `+${formatBytes(todayBytes)} today`,
              },
            },
          }));
        }
      },
      (err) => console.warn("[Firestore Telemetry Sync Error]", err)
    );

    return () => unsub();
  }, []);

  // 3. Telemetry Poller & Completion Watcher
  const pollTelemetry = useCallback(async () => {
    if (isPollingRef.current) return;
    isPollingRef.current = true;

    try {
      const nowTimeStr = new Date().toLocaleTimeString("en-US", {
        hour12: false,
        hour: "2-digit",
        minute: "2-digit",
        second: "2-digit",
      });

      if (!isOnline) {
        // When offline, preserve previous task list but label as stale
        const updatedHistory = [
          ...speedHistoryRef.current.slice(1),
          { time: nowTimeStr, speed: 0 },
        ];
        speedHistoryRef.current = updatedHistory;

        setSnapshot((prev) => ({
          ...prev,
          isStale: true,
          transferSpeedMbps: 0,
          speedHistory: updatedHistory,
          kpis: {
            ...prev.kpis,
            avgSpeed: { value: "0.00 MB/s", trend: "offline", isPositive: false },
          },
        }));
        completedFiredRef.current = false;
        return;
      }

      const data: BackendEngineStatus | null = await fetchBackendStatus();
      if (!data) {
        // Backend temporarily unreachable; preserve state and set stale flag
        setSnapshot((prev) => ({ ...prev, isStale: true }));
        return;
      }

      const statusStr = (data.status || "").toLowerCase();
      const isFinished =
        statusStr.includes("complete") ||
        statusStr.includes("finish") ||
        statusStr.includes("success") ||
        (data as any).completed === true;

      if (isFinished && !completedFiredRef.current) {
        completedFiredRef.current = true;
        commitTaskCompletionToFirestore(lastActiveTaskRef.current);
        if (onJobCompleted) onJobCompleted();
      }

      const tasks: ActiveTaskItem[] = [];
      let maxTaskSpeedMb = 0;

      if (Array.isArray(data.active_tasks) && data.active_tasks.length > 0) {
        data.active_tasks.forEach((t) => {
          const taskSpeedNum = parseSpeedStringToMbps(t.speed);
          if (taskSpeedNum > maxTaskSpeedMb) maxTaskSpeedMb = taskSpeedNum;

          tasks.push({
            id: t.id,
            name: t.name,
            source: t.type === "huggingface" ? "HuggingFace" : "Torrent",
            status: t.status === "transferring" ? "transferring" : "downloading",
            progress: Number(t.progress) || 0,
            speed: t.speed || "0.00 MB/s",
            eta: t.eta || "-",
          });
        });
      } else if (
        data.name &&
        (statusStr === "downloading" || statusStr === "transferring" || statusStr === "streaming")
      ) {
        // Support single-task root response formats
        const rootProgress = parseFloat(data.progress || "0") || 0;
        const rootSpeed = parseSpeedStringToMbps(data.speed || data.upload_speed);
        if (rootSpeed > maxTaskSpeedMb) maxTaskSpeedMb = rootSpeed;

        tasks.push({
          id: "active-stream",
          name: data.name,
          source: data.name.toLowerCase().includes("hf") ? "HuggingFace" : "Torrent",
          status: statusStr === "transferring" ? "transferring" : "downloading",
          progress: rootProgress,
          speed: data.speed || data.upload_speed || "0.00 MB/s",
          eta: data.upload_eta || "-",
        });
      }

      if (tasks.length > 0) {
        lastActiveTaskRef.current = tasks[0];
        completedFiredRef.current = false;
      }

      activeTasksCountRef.current = tasks.length;

      const downloadSpeedNum = parseSpeedStringToMbps(data.speed);
      const uploadSpeedNum = parseSpeedStringToMbps(data.upload_speed);
      const activeSpeedValue = Math.max(uploadSpeedNum, downloadSpeedNum, maxTaskSpeedMb);

      const nextHistory = [
        ...speedHistoryRef.current.slice(1),
        { time: nowTimeStr, speed: Number(activeSpeedValue.toFixed(2)) },
      ];
      speedHistoryRef.current = nextHistory;

      let displaySpeedString = "0.00 MB/s";
      if (tasks.length > 0 && activeSpeedValue > 0) {
        displaySpeedString = `${activeSpeedValue.toFixed(2)} MB/s`;
      }

      setSnapshot((prev) => ({
        ...prev,
        isStale: false,
        tasks,
        transferSpeedMbps: Number(activeSpeedValue.toFixed(2)),
        speedHistory: nextHistory,
        kpis: {
          ...prev.kpis,
          activeTransfers: {
            value: tasks.length,
            trend: tasks.length > 0 ? `+${tasks.length} running` : "idle",
            isPositive: tasks.length > 0,
          },
          avgSpeed: {
            value: displaySpeedString,
            trend: tasks.length > 0 ? "+active stream" : "0.00 MB/s",
            isPositive: tasks.length > 0,
          },
        },
      }));
    } finally {
      isPollingRef.current = false;
    }
  }, [isOnline, onJobCompleted]);

  // 4. Adaptive Polling Loop (Pauses/throttles when tab is backgrounded)
  useEffect(() => {
    let timeoutId: NodeJS.Timeout;
    let isMounted = true;

    const scheduleNextPoll = () => {
      const isHidden = typeof document !== "undefined" && document.visibilityState === "hidden";
      const hasWork = activeTasksCountRef.current > 0;

      // Adaptive interval: 1s while active, 3.5s while idle, 10s while offline, 15s while tab is hidden
      const delay = isHidden ? 15000 : !isOnline ? 10000 : hasWork ? 1000 : 3500;

      timeoutId = setTimeout(async () => {
        if (isMounted) {
          await pollTelemetry();
          if (isMounted) scheduleNextPoll();
        }
      }, delay);
    };

    const handleVisibilityChange = () => {
      if (document.visibilityState === "visible") {
        pollTelemetry();
      }
    };

    if (typeof document !== "undefined") {
      document.addEventListener("visibilitychange", handleVisibilityChange);
    }

    pollTelemetry().then(() => {
      if (isMounted) scheduleNextPoll();
    });

    return () => {
      isMounted = false;
      clearTimeout(timeoutId);
      if (typeof document !== "undefined") {
        document.removeEventListener("visibilitychange", handleVisibilityChange);
      }
    };
  }, [isOnline, pollTelemetry]);

  const triggerExtraction = async (req: EnqueueTaskRequest) => {
    completedFiredRef.current = false;
    const res = await enqueueExtraction(req);
    await pollTelemetry();
    return res;
  };

  const pauseTask = async (taskId: string) => {
    await sendTaskAction(taskId, "pause");
  };

  const resumeTask = async (taskId: string) => {
    await sendTaskAction(taskId, "resume");
  };

  const cancelTask = async (taskId: string) => {
    await sendTaskAction(taskId, "cancel");
    await pollTelemetry();
  };

  return {
    snapshot,
    firestoreBytes24h,
    firestoreTotalBytes,
    firestoreBytesToday,
    hourlyTransfers,
    triggerExtraction,
    pauseTask,
    resumeTask,
    cancelTask,
  };
}

/**
 * Task Management Commands
 * Note: The Codespace worker does not currently expose granular per-task pause/resume endpoints.
 * Returning unsupported rather than faking success, per roadmap Phase 5 specifications.
 */
export async function sendTaskAction(
  taskId: string,
  action: "pause" | "resume" | "cancel"
): Promise<{ status: "success" | "unsupported" | "error"; message?: string }> {
  if (action === "pause" || action === "resume") {
    // Truthful report: worker API does not implement per-task pause/resume
    return {
      status: "unsupported",
      message: `${action.toUpperCase()} is not currently supported by the active Codespace worker runtime.`,
    };
  }

  if (action === "cancel") {
    try {
      const authHeaders = await getAuthHeaders();
      const formData = new FormData();
      formData.append("task_id", taskId);

      const res = await fetch(`${BACKEND_PROXY_URL}/cancel`, {
        method: "POST",
        headers: authHeaders,
        body: formData,
      });

      if (res.ok) {
        return { status: "success" };
      }
      return { status: "error", message: `Cancel returned HTTP ${res.status}` };
    } catch (err: any) {
      return { status: "error", message: err.message || "Failed to cancel task" };
    }
  }

  return { status: "unsupported", message: "Action not supported" };
}

  }
}

  const raw = {
    NODE_ENV: process.env.NODE_ENV,
    BACKEND_URL: process.env.BACKEND_URL || process.env.NEXT_PUBLIC_BACKEND_URL,
    GDRIVE_CLIENT_EMAIL: process.env.GDRIVE_CLIENT_EMAIL,
    GDRIVE_PRIVATE_KEY: process.env.GDRIVE_PRIVATE_KEY,
    GDRIVE_FOLDER_ID: process.env.GDRIVE_FOLDER_ID || process.env.NEXT_PUBLIC_GDRIVE_FOLDER_ID,
    GITHUB_DISPATCH_TOKEN: process.env.GITHUB_DISPATCH_TOKEN || process.env.GH_TOKEN || process.env.GITHUB_TOKEN,
    GITHUB_REPO_OWNER: process.env.GITHUB_REPO_OWNER,
    GITHUB_REPO_NAME: process.env.GITHUB_REPO_NAME,
    ALLOWED_EMAILS: process.env.ALLOWED_EMAILS || process.env.NEXT_PUBLIC_ALLOWED_EMAILS,
    NEXT_PUBLIC_ALLOWED_EMAILS: process.env.NEXT_PUBLIC_ALLOWED_EMAILS,
    FIREBASE_PROJECT_ID: process.env.FIREBASE_PROJECT_ID,
  };

  const result = serverEnvSchema.safeParse(raw);
  if (!result.success) {

    .split(",")
    .map((e) => e.trim().toLowerCase())
    .filter(Boolean);
}

export async function triggerCodespaceWorkflow(): Promise<{ success: boolean; message: string }> {
  const env = getServerEnv();
  const token = env.GITHUB_DISPATCH_TOKEN;
  const owner = env.GITHUB_REPO_OWNER;
  const repo = env.GITHUB_REPO_NAME;

  if (!token) {
    return {
      success: false,
      message: "Server configuration missing: GITHUB_DISPATCH_TOKEN is not defined in environment.",
    };
  }

  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), 8000);

  try {
    const response = await fetch(
      `https://api.github.com/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}/dispatches`,
      {
        method: "POST",
        headers: {
          Authorization: `Bearer ${token}`,
          Accept: "application/vnd.github+json",
          "X-GitHub-Api-Version": "2022-11-28",
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          event_type: "trigger_pipeline",
        }),
        signal: controller.signal,
      }
    );

    // GitHub returns 204 No Content on a successful dispatch
    if (response.status === 204 || response.ok) {
      return {
        success: true,
        message: "Codespace wake signal dispatched successfully to GitHub Actions.",
      };
    }

    const errData = await response.json().catch(() => ({}));
    return {
      success: false,
      message: errData.message || `GitHub dispatch failed with HTTP ${response.status}`,
    };
  } catch (error: any) {
    return {
      success: false,
      message: error.name === "AbortError" ? "GitHub API request timed out" : "Network anomaly reaching GitHub API",
    };
  } finally {
    clearTimeout(timeoutId);
  }
}

/**
 * In-memory sliding window rate limiter for Next.js API routes
 * Prevents endpoint abuse, brute force, and rapid-fire wake/enqueue calls.
 */

interface RateLimitRecord {
  count: number;
  resetAt: number;
}

const rateLimitStore = new Map<string, RateLimitRecord>();

// Clean up stale entries every 5 minutes to prevent memory leaks
if (typeof setInterval !== "undefined") {
  setInterval(() => {
    const now = Date.now();
    for (const [key, record] of rateLimitStore.entries()) {
      if (now > record.resetAt) {
        rateLimitStore.delete(key);
      }
    }
  }, 5 * 60 * 1000).unref?.();
}

export interface RateLimitOptions {
  windowMs: number; // e.g. 60000 for 1 minute
  maxRequests: number; // e.g. 10 requests per minute
}

export function checkRateLimit(
  identifier: string,
  options: RateLimitOptions = { windowMs: 60 * 1000, maxRequests: 20 }
): { allowed: boolean; remaining: number; resetInMs: number } {
  const now = Date.now();
  const existing = rateLimitStore.get(identifier);

  if (!existing || now > existing.resetAt) {
    rateLimitStore.set(identifier, {
      count: 1,
      resetAt: now + options.windowMs,
    });
    return {
      allowed: true,
      remaining: options.maxRequests - 1,
      resetInMs: options.windowMs,
    };
  }

  if (existing.count >= options.maxRequests) {
    return {
      allowed: false,
      remaining: 0,
      resetInMs: Math.max(0, existing.resetAt - now),
    };
  }

  existing.count += 1;
  return {
    allowed: true,
    remaining: options.maxRequests - existing.count,
    resetInMs: Math.max(0, existing.resetAt - now),
  };
}

import { NextRequest, NextResponse } from "next/server";
import { getApps, initializeApp, getApp } from "firebase-admin/app";

export type UserRole = "admin" | "operator" | "viewer";

export interface AuthenticatedUser {

  role: UserRole;
}

export type AuthResult =
  | { authenticated: true; authorized: true; user: AuthenticatedUser }
  | { authenticated: false; authorized: false; response: NextResponse }
  | { authenticated: true; authorized: false; response: NextResponse };

function getFirebaseAdminAuth() {

/**
 * Server-side Authentication & Authorization Verifier
 * Verifies the incoming Firebase ID token against Google's public keys.
 * Enforces email whitelist and role requirements.
 */
export async function verifyServerAuth(
  req: NextRequest,
  options?: {
    requiredRole?: UserRole;
  }
): Promise<AuthResult> {
  const env = getServerEnv();
  const isDevOrTest = env.NODE_ENV === "development" || env.NODE_ENV === "test";

  // 1. Extract Bearer token from Authorization header or __session cookie
  let token: string | null = null;
  const authHeader = req.headers.get("authorization") || req.headers.get("Authorization");
  if (authHeader && authHeader.startsWith("Bearer ")) {
    token = authHeader.substring(7).trim();
  } else {
    token = req.cookies.get("__session")?.value || null;
  }

  // Allow test header strictly in development/test mode for automated tests
  if (isDevOrTest && !token && req.headers.get("x-test-auth-email")) {
    const testEmail = (req.headers.get("x-test-auth-email") || "").toLowerCase();
    const allowed = getAllowedEmails();
    if (allowed.length === 0 || allowed.includes(testEmail)) {
      return {
        authenticated: true,
        authorized: true,
        user: { uid: "test-user-id", email: testEmail, role: "admin" },
      };
    } else {
      return {
        authenticated: true,
        authorized: false,
        response: NextResponse.json(
          { error: "Forbidden: User is not authorized to access this resource" },
          { status: 403 }
        ),
      };
    }
  }

  if (!token) {
    return {
      authenticated: false,
      authorized: false,
      response: NextResponse.json(
        { error: "Unauthorized: Missing authentication token" },
        { status: 401 }
      ),
    };
  }

  try {
    const adminAuth = getFirebaseAdminAuth();
    const decodedToken = await adminAuth.verifyIdToken(token);
    const email = (decodedToken.email || "").toLowerCase();

    if (!email) {
      return {
        authenticated: false,
        authorized: false,
        response: NextResponse.json(
          { error: "Unauthorized: Token contains no verified email" },
          { status: 401 }
        ),
      };
    }

    // 2. Check Allowed Emails Whitelist
    const allowedEmails = getAllowedEmails();
    const isAllowed = allowedEmails.length === 0 || allowedEmails.includes(email);

    if (!isAllowed) {
      return {
        authenticated: true,
        authorized: false,
        response: NextResponse.json(
          { error: "Forbidden: Your account is not authorized to access RunSync" },
          { status: 403 }
        ),
      };
    }

    // First email in list is admin, others are operators
    const role: UserRole =
      allowedEmails.length > 0 && allowedEmails[0] === email ? "admin" : "operator";

    if (options?.requiredRole === "admin" && role !== "admin") {
      return {
        authenticated: true,
        authorized: false,
        response: NextResponse.json(
          { error: "Forbidden: Administrator privileges required for this action" },
          { status: 403 }
        ),
      };
    }

    return {
      authenticated: true,
      authorized: true,
      user: {
        uid: decodedToken.uid,
        email,
        role,
      },
    };
  } catch (error: any) {
    console.warn("[Server Auth] Token verification rejected:", error?.message || error);
    return {
      authenticated: false,
      authorized: false,
      response: NextResponse.json(
        { error: "Unauthorized: Invalid or expired authentication token" },
        { status: 401 }
      ),
    };
  }
}

/**
 * Telemetry formatting and unit conversion utilities
 */

export function formatBytes(bytes: number, decimals: number = 1): string {
  if (!bytes || bytes <= 0 || isNaN(bytes)) return "0 B";
  const k = 1024;
  const dm = decimals < 0 ? 0 : decimals;
  const sizes = ["B", "KB", "MB", "GB", "TB", "PB"];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  const safeIndex = Math.min(Math.max(i, 0), sizes.length - 1);
  return `${parseFloat((bytes / Math.pow(k, safeIndex)).toFixed(dm))} ${sizes[safeIndex]}`;
}

export function parseSpeedStringToMbps(speedStr: string | undefined): number {
  if (!speedStr) return 0;
  const cleaned = speedStr.trim().toLowerCase();
  if (
    cleaned.includes("finish") ||
    cleaned.includes("idle") ||
    cleaned.includes("done") ||
    cleaned.includes("offline") ||
    cleaned.includes("calculating")
  ) {
    return 0;
  }

  const numericMatch = cleaned.match(/([\d.]+)/);
  if (!numericMatch) return 0;

  const rawVal = parseFloat(numericMatch[1]);
  if (isNaN(rawVal)) return 0;

  if (cleaned.includes("gb/s") || cleaned.includes("gib/s") || cleaned.includes("gbyte/s")) {
    return rawVal * 1024;
  }
  if (cleaned.includes("mb/s") || cleaned.includes("mib/s") || cleaned.includes("mbyte/s")) {
    return rawVal;
  }
  if (cleaned.includes("kb/s") || cleaned.includes("kib/s") || cleaned.includes("kbyte/s")) {
    return rawVal / 1024;
  }
  if (cleaned.includes("b/s") || cleaned.includes("byte/s")) {
    return rawVal / (1024 * 1024);
  }
  return rawVal;
}

export function parseBytesFromTaskName(name: string): number {
  if (!name) return 0;
  const match = name.match(/(\d+(?:\.\d+)?)\s*(gb|gib|mb|mib|kb|kib)/i);
  if (!match) return 0;

  const val = parseFloat(match[1]);
  const unit = match[2].toLowerCase();

  if (unit.startsWith("g")) return Math.round(val * 1024 * 1024 * 1024);
  if (unit.startsWith("m")) return Math.round(val * 1024 * 1024);
  if (unit.startsWith("k")) return Math.round(val * 1024);
  return 0;
}

  const k = 1024;

/**
 * Formats seconds into a human-readable ETA string (e.g., "42m", "12m 42s")
 */
export function formatEta(seconds: number): string {
  if (!seconds || seconds <= 0 || !isFinite(seconds)) return "-";
  const s = Math.floor(seconds);
  const minutes = Math.floor(s / 60);
  const hours = Math.floor(minutes / 60);

  if (hours > 0) {
    const remainingMins = minutes % 60;
    return `${hours}h ${remainingMins}m`;
  }
  if (minutes > 0) {
    const remainingSecs = s % 60;
    return remainingSecs > 0 ? `${minutes}m ${remainingSecs}s` : `${minutes}m`;
  }
  return `${s}s`;
}

/**
 * RunSync Input Validation & Normalization Library
 * Validates extraction inputs (Hugging Face URLs/IDs, AcademicTorrents, Magnets, Infohashes)
 * Prevents invalid/malicious payloads from reaching server endpoints or client fetchers.
 */

export type DetectedSource = "huggingface" | "academic_torrent";

export interface ExtractionValidationResult {
  valid: boolean;
  error?: string;
  detectedSource?: DetectedSource;
  normalizedInput?: string;
  isDataset?: boolean;
  repoId?: string;
}

export function validateExtractionInput(
  raw: string,
  forcedBadge?: "auto" | "hf" | "academic"
): ExtractionValidationResult {
  if (!raw || typeof raw !== "string") {
    return { valid: false, error: "Input cannot be empty." };
  }

  const trimmed = raw.trim();
  if (trimmed.length === 0) {
    return { valid: false, error: "Input cannot be empty." };
  }

  if (trimmed.length > 2048) {
    return {
      valid: false,
      error: "Input exceeds maximum allowed length of 2048 characters.",
    };
  }

  // Reject malicious/dangerous schemes
  const lower = trimmed.toLowerCase();
  if (
    lower.startsWith("javascript:") ||
    lower.startsWith("data:") ||
    lower.startsWith("vbscript:") ||
    lower.startsWith("file:")
  ) {
    return { valid: false, error: "Invalid URL scheme." };
  }

  // 1. Magnet link check
  if (lower.startsWith("magnet:?")) {
    if (!lower.includes("xt=urn:btih:")) {
      return {
        valid: false,
        error: "Magnet link must contain a valid BitTorrent infohash (xt=urn:btih:).",
      };
    }
    return {
      valid: true,
      detectedSource: "academic_torrent",
      normalizedInput: trimmed,
    };
  }

  // 2. Infohash (40-char hex string or 32-char base32 string)
  if (/^[0-9a-fA-F]{40}$/.test(trimmed)) {
    return {
      valid: true,
      detectedSource: "academic_torrent",
      normalizedInput: `magnet:?xt=urn:btih:${trimmed}`,
    };
  }
  if (/^[2-7a-zA-Z]{32}$/.test(trimmed)) {
    return {
      valid: true,
      detectedSource: "academic_torrent",
      normalizedInput: `magnet:?xt=urn:btih:${trimmed}`,
    };
  }

  // 3. Academic Torrents URL
  if (lower.includes("academictorrents.com")) {
    const match = trimmed.match(/\/details\/([0-9a-fA-F]{40})/i);
    if (match) {
      return {
        valid: true,
        detectedSource: "academic_torrent",
        normalizedInput: `magnet:?xt=urn:btih:${match[1]}`,
      };
    }
    return {
      valid: true,
      detectedSource: "academic_torrent",
      normalizedInput: trimmed,
    };
  }

  // 4. Direct .torrent URL
  if (lower.endsWith(".torrent") || lower.includes(".torrent?")) {
    if (!lower.startsWith("http://") && !lower.startsWith("https://")) {
      return {
        valid: false,
        error: "Torrent URL must begin with https:// or http://.",
      };
    }
    return {
      valid: true,
      detectedSource: "academic_torrent",
      normalizedInput: trimmed,
    };
  }

  // If user explicitly forced academic torrents but didn't supply a valid torrent format:
  if (forcedBadge === "academic") {
    return {
      valid: false,
      error: "Input is not a recognized magnet link, infohash, or .torrent URL.",
    };
  }

  // 5. Hugging Face repository validation
  const isDataset =
    lower.includes("/datasets/") ||
    lower.startsWith("datasets/") ||
    (forcedBadge === "hf" && lower.startsWith("datasets/"));

  let cleanId = trimmed;

  if (lower.startsWith("https://huggingface.co/") || lower.startsWith("http://huggingface.co/")) {
    cleanId = trimmed
      .replace(/^https?:\/\/huggingface\.co\/(datasets\/)?/, "")
      .split("/tree/")[0]
      .split("/blob/")[0]
      .split("/resolve/")[0]
      .trim();
  } else if (lower.startsWith("hf.co/")) {
    cleanId = trimmed
      .replace(/^hf\.co\/(datasets\/)?/, "")
      .split("/tree/")[0]
      .split("/blob/")[0]
      .trim();
  }

  // Remove trailing slashes
  cleanId = cleanId.replace(/\/+$/, "");

  // Hugging Face model/dataset repos must follow [namespace]/[model_name]
  const repoRegex = /^[A-Za-z0-9_.-]+\/[A-Za-z0-9_.-]+$/;
  if (!repoRegex.test(cleanId)) {
    return {
      valid: false,
      error:
        "Please enter a valid Hugging Face repository (e.g. 'org/model-name' or full URL) or a BitTorrent magnet link.",
    };
  }

  return {
    valid: true,
    detectedSource: "huggingface",
    normalizedInput: isDataset
      ? `https://huggingface.co/datasets/${cleanId}`
      : `https://huggingface.co/${cleanId}`,
    isDataset,
    repoId: cleanId,
  };
}
