import { NextRequest, NextResponse } from "next/server";

export const dynamic = "force-dynamic";

const NGROK_BACKEND_URL =
  process.env.BACKEND_URL ||
  process.env.NEXT_PUBLIC_BACKEND_URL ||
  "https://worrier-verbose-blustery.ngrok-free.dev";

function buildTargetUrl(rawPath: string[], search: string): string {
  // Prevent duplicate /api/v1 prefix if already supplied
  let cleanSegments = [...rawPath];
  if (cleanSegments[0] === "api" && cleanSegments[1] === "v1") {
    cleanSegments = cleanSegments.slice(2);
  }
  const cleanPath = cleanSegments.join("/");
  return `${NGROK_BACKEND_URL}/api/v1/${cleanPath}${search}`;
}

export async function GET(
  req: NextRequest,
  context: { params: Promise<{ path: string[] }> }
) {
  const { path } = await context.params;
  const search = req.nextUrl.search || "";
  const targetUrl = buildTargetUrl(path, search);

  try {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 7000);

    const res = await fetch(targetUrl, {
      method: "GET",
      headers: {
        "ngrok-skip-browser-warning": "69420",
        Accept: "application/json",
      },
      signal: controller.signal,
      cache: "no-store",
    });

    clearTimeout(timeoutId);

    if (!res.ok) {
      return NextResponse.json(
        { status: "standby", message: `Remote worker returned HTTP ${res.status}` },
        { status: 200 }
      );
    }

    const data = await res.json();
    return NextResponse.json(data);
  } catch {
    // Return clean fallback so Next.js frontend state remains smooth
    return NextResponse.json(
      { status: "standby", cluster: "Codespace • Standby" },
      { status: 200 }
    );
  }
}

export async function POST(
  req: NextRequest,
  context: { params: Promise<{ path: string[] }> }
) {
  const { path } = await context.params;
  const search = req.nextUrl.search || "";
  const targetUrl = buildTargetUrl(path, search);

  try {
    const contentType = req.headers.get("content-type") || "";
    const headers: Record<string, string> = {
      "ngrok-skip-browser-warning": "69420",
    };

    let bodyPayload: any;

    if (
      contentType.includes("multipart/form-data") ||
      contentType.includes("application/x-www-form-urlencoded")
    ) {
      // Fetch automatically calculates multipart boundary when given FormData
      bodyPayload = await req.formData();
    } else if (contentType.includes("application/json")) {
      bodyPayload = await req.text();
      headers["Content-Type"] = "application/json";
    } else {
      bodyPayload = await req.text();
    }

    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 12000);

    const res = await fetch(targetUrl, {
      method: "POST",
      headers,
      body: bodyPayload,
      signal: controller.signal,
      cache: "no-store",
    });

    clearTimeout(timeoutId);

    const data = await res.json();
    return NextResponse.json(data);
  } catch (err: any) {
    return NextResponse.json(
      {
        status: "error",
        message: err.message || "Failed to reach remote Codespace engine",
      },
      { status: 502 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import { google } from "googleapis";
import { Readable } from "stream";

export const dynamic = "force-dynamic";

function cleanPrivateKey(key?: string): string {
  if (!key) return "";
  let cleaned = key.trim();
  if (
    (cleaned.startsWith('"') && cleaned.endsWith('"')) ||
    (cleaned.startsWith("'") && cleaned.endsWith("'"))
  ) {
    cleaned = cleaned.slice(1, -1);
  }
  return cleaned.replace(/\\n/g, "\n").replace(/\r/g, "").trim();
}

export async function GET(req: NextRequest) {
  try {
    const { searchParams } = new URL(req.url);
    const fileId = searchParams.get("fileId");

    if (!fileId) {
      return NextResponse.json({ error: "Missing fileId parameter" }, { status: 400 });
    }

    const clientEmail = process.env.GDRIVE_CLIENT_EMAIL;
    const rawKey = process.env.GDRIVE_PRIVATE_KEY;

    if (!clientEmail || !rawKey) {
      return NextResponse.json(
        { error: "Google Drive service credentials not configured" },
        { status: 500 }
      );
    }

    const auth = new google.auth.JWT({
      email: clientEmail,
      key: cleanPrivateKey(rawKey),
      scopes: ["https://www.googleapis.com/auth/drive.readonly"],
    });

    const drive = google.drive({ version: "v3", auth });

    // Fetch file metadata for content type and accurate size
    const metadata = await drive.files.get({
      fileId,
      fields: "name, mimeType, size",
      supportsAllDrives: true,
    });

    const driveRes = await drive.files.get(
      { fileId, alt: "media", supportsAllDrives: true },
      { responseType: "stream" }
    );

    const headers = new Headers();
    const fileName = metadata.data.name || "download";
    headers.set(
      "Content-Disposition",
      `attachment; filename="${encodeURIComponent(fileName)}"`
    );
    headers.set(
      "Content-Type",
      metadata.data.mimeType || "application/octet-stream"
    );

    if (metadata.data.size) {
      headers.set("Content-Length", metadata.data.size);
    }

    // Convert Node.js Readable stream into standard Web ReadableStream
    const webStream = Readable.toWeb(driveRes.data as unknown as Readable);

    return new Response(webStream as BodyInit, { headers });
  } catch (error: any) {
    console.error("[Drive Download Error]", error);
    return NextResponse.json(
      { error: error.message || "Failed to download file from Google Drive" },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import { google } from "googleapis";

export const dynamic = "force-dynamic";

function cleanPrivateKey(key?: string): string {
  if (!key) return "";
  let cleaned = key.trim();
  if (
    (cleaned.startsWith('"') && cleaned.endsWith('"')) ||
    (cleaned.startsWith("'") && cleaned.endsWith("'"))
  ) {
    cleaned = cleaned.slice(1, -1);
  }
  return cleaned.replace(/\\n/g, "\n").replace(/\r/g, "").trim();
}

function getDriveClient() {
  const clientEmail = process.env.GDRIVE_CLIENT_EMAIL;
  const rawPrivateKey = process.env.GDRIVE_PRIVATE_KEY;
  const privateKey = cleanPrivateKey(rawPrivateKey);

  if (!clientEmail || !privateKey) {
    throw new Error(
      "Missing Google Drive credentials in .env.local (GDRIVE_CLIENT_EMAIL / GDRIVE_PRIVATE_KEY)"
    );
  }

  const auth = new google.auth.JWT({
    email: clientEmail,
    key: privateKey,
    scopes: ["https://www.googleapis.com/auth/drive.readonly"],
  });

  return google.drive({ version: "v3", auth });
}

export async function GET(req: NextRequest) {
  try {
    const { searchParams } = new URL(req.url);
    const targetFolderId =
      searchParams.get("folderId") ||
      process.env.GDRIVE_FOLDER_ID ||
      process.env.NEXT_PUBLIC_GDRIVE_FOLDER_ID ||
      "1W5rVvmzDA5hpFEH8Sd53IyLb9R-bmY9O";

    const search = searchParams.get("search");
    const drive = getDriveClient();

    const allItemsRes = await drive.files.list({
      q: "trashed = false",
      fields: "files(id, name, mimeType, size, parents, modifiedTime, webViewLink, iconLink)",
      pageSize: 1000,
      supportsAllDrives: true,
      includeItemsFromAllDrives: true,
    });

    const allItems = allItemsRes.data.files || [];

    const now = Date.now();
    const twentyFourHoursAgo = now - 24 * 60 * 60 * 1000;
    const startOfToday = new Date();
    startOfToday.setHours(0, 0, 0, 0);
    const startOfTodayMs = startOfToday.getTime();

    let bytes24h = 0;
    let bytesToday = 0;

    const parentMap = new Map<string, string[]>();
    for (const item of allItems) {
      if (item.id && item.parents) {
        parentMap.set(item.id, item.parents);
      }
    }

    const folderSizeMap = new Map<string, number>();
    let totalStorageBytes = 0;

    for (const item of allItems) {
      const isFolder = item.mimeType === "application/vnd.google-apps.folder";
      if (!isFolder && item.size) {
        const bytes = parseInt(item.size, 10) || 0;
        totalStorageBytes += bytes;

        const modTime = item.modifiedTime ? new Date(item.modifiedTime).getTime() : 0;
        if (modTime >= twentyFourHoursAgo) {
          bytes24h += bytes;
        }
        if (modTime >= startOfTodayMs) {
          bytesToday += bytes;
        }

        // Bubble up bytes to ancestor folders
        let queue = item.parents ? [...item.parents] : [];
        const visited = new Set<string>();

        while (queue.length > 0) {
          const parentId = queue.shift()!;
          if (visited.has(parentId)) continue;
          visited.add(parentId);

          folderSizeMap.set(
            parentId,
            (folderSizeMap.get(parentId) || 0) + bytes
          );

          const grandParents = parentMap.get(parentId);
          if (grandParents) {
            for (const gp of grandParents) {
              if (!visited.has(gp)) queue.push(gp);
            }
          }
        }
      }
    }

    let displayItems: any[] = [];
    if (search && search.trim()) {
      const term = search.trim().toLowerCase();
      displayItems = allItems.filter((item) =>
        (item.name || "").toLowerCase().includes(term)
      );
    } else {
      displayItems = allItems.filter(
        (item) => item.parents && item.parents.includes(targetFolderId)
      );
    }

    const files = displayItems.map((file) => {
      const isFolder = file.mimeType === "application/vnd.google-apps.folder";
      const sizeBytes = isFolder
        ? folderSizeMap.get(file.id || "") || 0
        : file.size
        ? parseInt(file.size, 10)
        : 0;

      return {
        id: file.id || "",
        name: file.name || "Untitled",
        mimeType: file.mimeType || "",
        isFolder,
        sizeBytes,
        modifiedTime: file.modifiedTime || new Date().toISOString(),
        webViewLink: file.webViewLink || "",
        iconLink: file.iconLink || "",
      };
    });

    const folderStorage = folderSizeMap.get(targetFolderId);

    // Direct items inside CodespaceDownloads (each folder or file is 1 distinct extraction task)
    const directChildrenCount = displayItems.length;

    // True count of completed extraction tasks (baseline 135)
    const totalCompletedCount = Math.max(directChildrenCount, 135);

    return NextResponse.json({
      success: true,
      folderId: targetFolderId,
      totalStorageBytes: typeof folderStorage === "number" ? folderStorage : totalStorageBytes,
      bytes24h,
      bytesToday,
      totalCompletedCount,
      files,
    });
  } catch (error: any) {
    console.error("[Drive API Error]", error);
    return NextResponse.json(
      { error: error.message || "Failed to query Google Drive API" },
      { status: 500 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";
import {
  doc,
  getDoc,
  setDoc,
  updateDoc,
  increment,
  serverTimestamp,
} from "firebase/firestore";
import { db } from "@/lib/firebase";

export async function GET() {
  try {
    const docRef = doc(db, "analytics", "dashboard");
    const snap = await getDoc(docRef);

    if (!snap.exists()) {
      return NextResponse.json({
        completed_count: 135,
        total_bytes_extracted: 122514567168,
        hourly_transfers: {},
      });
    }

    return NextResponse.json(snap.data());
  } catch (error: any) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

export async function POST(req: NextRequest) {
  try {
    const body = await req.json().catch(() => ({}));
    const bytes = Number(body.bytes) || 0;

    const docRef = doc(db, "analytics", "dashboard");
    const currentHour = new Date().getHours().toString().padStart(2, "0");

    try {
      await updateDoc(docRef, {
        completed_count: increment(1),
        total_bytes_extracted: increment(bytes),
        [`hourly_transfers.${currentHour}`]: increment(bytes),
        last_updated: serverTimestamp(),
      });
    } catch (err: any) {
      if (err.code === "not-found") {
        await setDoc(docRef, {
          completed_count: 135 + 1,
          total_bytes_extracted: 122514567168 + bytes,
          hourly_transfers: { [currentHour]: bytes },
          last_updated: serverTimestamp(),
        });
      } else {
        throw err;
      }
    }

    return NextResponse.json({ success: true });
  } catch (error: any) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

        showToast(result.message || "Task registered with the worker engine.", "info", 4000);
      }
    } catch (err: any) {
      showToast(`Dispatch failed: ${err.message || "Network unreachable"}`, "warning", 6000);
    } finally {
      setIsSubmitting(false);
      pendingExtractionRef.current = null;
    }
  };

  const handleExtractSubmit = async (payload: {
    url: string;
    source: SourceBadge;
    mode: ExecutionMode;
  }) => {
    const targetUrl = payload.url.trim();
    if (!targetUrl) return;

    const detectedEngine = detectPayloadType(targetUrl, payload.source);

    if (payload.mode === "local") {
      await handleLocalBrowserFetch(targetUrl, detectedEngine);
      return;
    }

    if (payload.mode === "cloud" && !isOnline) {
      pendingExtractionRef.current = payload;
      setShowWakeModal(true);
      return;
    }

    await dispatchExtractionPayload(payload);
  };

  const handleConfirmWake = async () => {
    setShowWakeModal(false);
    showToast("Boot signal dispatched. Spawning Codespace container...", "info", 0);

    try {
      await wakeUpCodespace();

      let attempts = 0;
      let backendReady = false;

      while (attempts < 20) {
        await new Promise((r) => setTimeout(r, 2500));
        const alive = await checkStatus();
        if (alive) {
          backendReady = true;
          break;
        }
        attempts++;
      }

      if (backendReady) {
        showToast("Codespace is online! Tunnel established.", "success", 3000);
        if (pendingExtractionRef.current) {
          await dispatchExtractionPayload(pendingExtractionRef.current);
        }
      } else {
        showToast("Boot timed out. Check GitHub Actions workflow runs.", "warning", 6000);
        pendingExtractionRef.current = null;
      }
    } catch (err: any) {
      showToast(`Wake sequence failed: ${err.message || "Error"}`, "warning", 6000);
      pendingExtractionRef.current = null;
    }
  };

  return (
    <div className="space-y-4 md:space-y-5 pb-8 sm:pb-6 relative">
      <ServerWakeModal
        isOpen={showWakeModal}
        isBooting={isBooting}
        onClose={() => {
          setShowWakeModal(false);
          pendingExtractionRef.current = null;
        }}
        onConfirmWake={handleConfirmWake}
      />

      {toast.show && (
        <div className="fixed bottom-20 sm:bottom-6 right-4 sm:right-6 z-50 flex items-center gap-3 px-4 py-3 rounded-xl bg-[#0D1424] border border-[#1E2C48] text-xs shadow-2xl text-slate-200 animate-in fade-in slide-in-from-bottom-3 duration-200 max-w-md">
          {toast.type === "success" && <CheckCircle2 className="w-4 h-4 text-emerald-400 shrink-0" />}
          {toast.type === "warning" && <AlertTriangle className="w-4 h-4 text-amber-400 shrink-0" />}
          {toast.type === "info" && <Loader2 className="w-4 h-4 text-blue-400 animate-spin shrink-0" />}
          <span className="flex-1 leading-normal">{toast.message}</span>
          <button
            onClick={() => setToast((prev) => ({ ...prev, show: false }))}
            className="text-slate-500 hover:text-slate-300 ml-1"
          >
            <X className="w-3.5 h-3.5" />
          </button>
        </div>
      )}

      {/* Row 1: Greeting */}
      <WelcomeBanner
        userName="Saiteja"
        isOnline={isOnline}
        isBooting={isBooting}
        cluster={cluster}
        onWake={wakeUpCodespace}
      />

      {/* Row 2: KPI Metrics Cards (All values & sub-trends bound to Firestore) */}
      <MetricsRow
        metrics={snapshot.kpis}
        liveExtractedBytes={firestoreTotalBytes}
        liveBytesToday={firestoreBytesToday}
      />

        <AnalyticsGrid
          speedHistory={snapshot.speedHistory}
          currentSpeed={String(snapshot.kpis.avgSpeed.value)}
          bytes24h={firestoreBytes24h}
        />
        <StorageCard

            usedBytes: liveStorageBytes,
            totalBytes: 5 * 1024 * 1024 * 1024 * 1024,
          }}
        />
      </div>

      {/* Row 5: Operations List */}
      <div className="w-full">
        <ActiveTransfers
          tasks={snapshot.tasks}
          onPause={pauseTask}
          onResume={resumeTask}
          onCancel={cancelTask}
        />
      </div>

function getFileIcon(name: string, mimeType: string) {
  const lower = name.toLowerCase();

  if (
    lower.endsWith(".safetensors") ||
    lower.endsWith(".bin") ||
    lower.endsWith(".pt") ||
    lower.endsWith(".pth") ||
    lower.endsWith(".onnx") ||
    lower.endsWith(".gguf")
  ) {
    return <Binary className="w-4 h-4 text-purple-400 shrink-0" />;
  }

  if (
    lower.endsWith(".parquet") ||
    lower.endsWith(".arrow") ||
    lower.endsWith(".csv") ||
    lower.endsWith(".json") ||
    lower.endsWith(".jsonl")
  ) {
    return <Database className="w-4 h-4 text-emerald-400 shrink-0" />;
  }

  if (
    lower.endsWith(".mkv") ||
    lower.endsWith(".mp4") ||
    lower.endsWith(".avi") ||
    lower.endsWith(".mov") ||
    mimeType.startsWith("video/")
  ) {
    return <Film className="w-4 h-4 text-amber-400 shrink-0" />;
  }

  if (
    lower.endsWith(".zip") ||
    lower.endsWith(".tar") ||
    lower.endsWith(".gz") ||
    lower.endsWith(".7z") ||
    lower.endsWith(".rar")
  ) {
    return <FileArchive className="w-4 h-4 text-rose-400 shrink-0" />;
  }

  if (
    lower.endsWith(".py") ||
    lower.endsWith(".ts") ||
    lower.endsWith(".js") ||
    lower.endsWith(".sh") ||
    lower.endsWith(".yaml") ||
    lower.endsWith(".yml")
  ) {
    return <FileCode className="w-4 h-4 text-cyan-400 shrink-0" />;
  }

  return <FileText className="w-4 h-4 text-slate-400 shrink-0" />;
}

import React, { useState, useEffect } from "react";
import {
  signInWithPopup,

  onAuthStateChanged,
  User,
} from "firebase/auth";
import { auth, googleProvider } from "@/lib/firebase";
import { ShieldAlert, LogIn, Loader2, Zap } from "lucide-react";

export function AuthGate({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState<boolean>(true);
  const [accessDenied, setAccessDenied] = useState<string | null>(null);

  const allowedEmails = (process.env.NEXT_PUBLIC_ALLOWED_EMAILS || "")
    .split(",")
    .map((e) => e.trim().toLowerCase())
    .filter(Boolean);

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, (currentUser) => {
      if (currentUser && currentUser.email) {
        const email = currentUser.email.toLowerCase();
        if (allowedEmails.includes(email)) {
          setUser(currentUser);
          setAccessDenied(null);
        } else {
          signOut(auth);
          setUser(null);
          setAccessDenied(currentUser.email);
        }
      } else {
        setUser(null);
      }
      setLoading(false);
    });

    return () => unsubscribe();
  }, []);

  const handleGoogleLogin = async () => {
    setLoading(true);
    setAccessDenied(null);
    try {
      await signInWithPopup(auth, googleProvider);
    } catch (err: any) {
      console.error("Login failed:", err);
      setLoading(false);
    }
  };

  if (loading) {
    return (
      <div className="h-screen w-screen bg-[#080C14] flex flex-col items-center justify-center text-slate-400 gap-3">
        <Loader2 className="w-8 h-8 text-blue-500 animate-spin" />
        <span className="text-xs font-mono">Verifying authorization...</span>
      </div>
    );
  }

  if (!user) {
    return (
      <div className="h-screen w-screen bg-[#080C14] flex flex-col items-center justify-center p-4">
        <div className="w-full max-w-sm p-6 sm:p-8 rounded-2xl bg-[#0D1424] border border-[#1A263D] shadow-2xl flex flex-col items-center text-center space-y-5">
          <div className="w-12 h-12 rounded-2xl bg-gradient-to-tr from-blue-600 to-indigo-500 flex items-center justify-center text-white shadow-lg shadow-blue-500/30">
            <Zap className="w-6 h-6 fill-current" />
          </div>

          <div>
            <h1 className="text-xl font-bold text-white tracking-tight">RunSync 2.0</h1>
            <p className="text-xs text-slate-400 mt-1">
              Private Cloud Extraction & Swarm Pipeline
            </p>
          </div>

          {accessDenied && (
            <div className="w-full p-3 rounded-xl bg-rose-500/10 border border-rose-500/30 text-rose-300 text-xs flex items-center gap-2.5 text-left">
              <ShieldAlert className="w-4 h-4 shrink-0 text-rose-400" />
              <span>
                Access Denied for <strong>{accessDenied}</strong>. Email is not whitelisted.
              </span>

import React from "react";
import { cn } from "@/lib/utils";

export type SparklineType = "line" | "bars";
export type SparklineColor = "blue" | "emerald" | "purple" | "amber";

interface SparklineProps {
  type?: SparklineType;
  color?: SparklineColor;
  data?: number[];
  className?: string;
  width?: number;
  height?: number;
}

export const Sparkline: React.FC<SparklineProps> = ({
  type = "line",
  color = "blue",
  data = [12, 18, 14, 25, 20, 28, 22, 34, 30],
  className,
  width = 96,
  height = 36,
}) => {
  const colorMap = {
    blue: {
      stroke: "#38BDF8",
      fill: "#0284C7",
      bar: "fill-blue-500",
      glow: "drop-shadow(0 0 4px rgba(56,189,248,0.5))",
    },
    emerald: {
      stroke: "#34D399",
      fill: "#059669",
      bar: "fill-emerald-500",
      glow: "drop-shadow(0 0 4px rgba(52,211,153,0.5))",
    },
    purple: {
      stroke: "#C084FC",
      fill: "#7E22CE",
      bar: "fill-purple-500",
      glow: "drop-shadow(0 0 4px rgba(192,132,252,0.5))",
    },
    amber: {
      stroke: "#FBBF24",
      fill: "#D97706",
      bar: "fill-amber-500",
      glow: "drop-shadow(0 0 4px rgba(251,191,36,0.5))",
    },
  };

  const selectedColor = colorMap[color];
  const safeData = data.length > 0 ? data : [0];

  // Renders vertical micro-bars (Completed & Extracted KPIs)
  if (type === "bars") {
    const maxVal = Math.max(...safeData, 1);
    const barWidth = 4;
    const gap = 4;
    const totalBars = safeData.length;
    const svgWidth = totalBars * (barWidth + gap);

    return (
      <svg
        width={svgWidth}
        height={height}
        viewBox={`0 0 ${svgWidth} ${height}`}
        className={cn("overflow-visible", className)}
      >
        {safeData.map((val, index) => {
          const barHeight = Math.max((val / maxVal) * (height - 6), 3);
          const x = index * (barWidth + gap);
          const y = height - barHeight;
          const opacity = 0.4 + (index / totalBars) * 0.6;

          return (
            <rect
              key={index}
              x={x}
              y={y}
              width={barWidth}
              height={barHeight}
              rx={2}
              className={selectedColor.bar}
              style={{ opacity }}
            />
          );
        })}
      </svg>
    );
  }

  // Renders smooth SVG wave curves (Active Transfers & Speed KPIs)
  const min = Math.min(...safeData);
  const max = Math.max(...safeData);
  const range = max - min || 1;
  const padding = 4;
  const usableHeight = height - padding * 2;

  // Safe point normalization (prevents divide-by-zero on single-item data)
  const safeDivisor = Math.max(safeData.length - 1, 1);

  const points = safeData.map((val, i) => {
    const x = (i / safeDivisor) * (width - 8) + 4;
    const y = height - padding - ((val - min) / range) * usableHeight;
    return { x, y };
  });

  let pathD = `M ${points[0].x} ${points[0].y}`;
  if (points.length === 1) {
    pathD += ` L ${width - 4} ${points[0].y}`;
  } else {
    for (let i = 0; i < points.length - 1; i++) {
      const curr = points[i];
      const next = points[i + 1];
      const cpX = (curr.x + next.x) / 2;
      pathD += ` C ${cpX} ${curr.y}, ${cpX} ${next.y}, ${next.x} ${next.y}`;
    }
  }

  return (
    <svg
      width={width}
      height={height}
      viewBox={`0 0 ${width} ${height}`}
      className={cn("overflow-visible", className)}
    >
      <path
        d={pathD}
        fill="none"
        stroke={selectedColor.stroke}
        strokeWidth={2.2}
        strokeLinecap="round"
        style={{ filter: selectedColor.glow }}
      />
    </svg>
  );
};

  const [cluster, setCluster] = useState<string>("Codespace • US-West");
  const [isBooting, setIsBooting] = useState<boolean>(false);
  const [engineState, setEngineState] = useState<string>("idle");

  const isBootingRef = useRef(isBooting);
  isBootingRef.current = isBooting;

  const activePollTimerRef = useRef<NodeJS.Timeout | null>(null);

  const checkStatus = useCallback(async (): Promise<boolean> => {
    const health = await checkBackendHealth();
    if (health && health.status === "online") {
      setStatus("online");
      setIsBooting(false);
      if (health.cluster) setCluster(health.cluster);
      if (health.engine) setEngineState(health.engine);
      return true;
    } else {
      if (!isBootingRef.current) {
        setStatus("standby");
        setEngineState("idle");
      }
      return false;
    }
  }, []);

  // Main polling loop
  useEffect(() => {
    checkStatus();

    const interval = setInterval(async () => {
      await checkStatus();
    }, isBooting ? 2500 : pollIntervalMs);

    return () => clearInterval(interval);
  }, [checkStatus, pollIntervalMs, isBooting]);

  // Clean up any active boot poll timer on unmount
  useEffect(() => {
    return () => {
      if (activePollTimerRef.current) {
        clearInterval(activePollTimerRef.current);
      }
    };
  }, []);

  /**
   * Dispatches GitHub Action to wake up Codespace and polls until alive
   */
  const wakeUpCodespace = useCallback(async (): Promise<boolean> => {
    if (activePollTimerRef.current) {
      clearInterval(activePollTimerRef.current);
    }

    setIsBooting(true);
    setStatus("booting");

    const dispatched = await wakeCodespaceInstance();
    if (!dispatched) {
      setIsBooting(false);
      setStatus("standby");
      return false;
    }

    // Actively poll for up to 90 seconds (Codespace boot + Ngrok handshake takes ~25-40s)
    let attempts = 0;
    const maxAttempts = 30; // 30 * 3s = 90s

    activePollTimerRef.current = setInterval(async () => {
      attempts++;
      const isAlive = await checkStatus();
      if (isAlive || attempts >= maxAttempts) {
        if (activePollTimerRef.current) {
          clearInterval(activePollTimerRef.current);
          activePollTimerRef.current = null;
        }
        if (!isAlive) {
          setIsBooting(false);
          setStatus("standby");
        }
      }
    }, 3000);

    return true;
  }, [checkStatus]);

  return {
    status,
    isOnline: status === "online",
    isBooting,
    cluster,
    engineState,
    checkStatus,
    wakeUpCodespace,
  };
}

function parseBytesFromTaskName(name: string): number {
  if (!name) return 500 * 1024 * 1024;
  const match = name.match(/(\d+(?:\.\d+)?)\s*(gb|gib|mb|mib|kb|kib)/i);
  if (!match) return 500 * 1024 * 1024;

  const val = parseFloat(match[1]);
  const unit = match[2].toLowerCase();

  if (unit.startsWith("g")) return Math.round(val * 1024 * 1024 * 1024);
  if (unit.startsWith("m")) return Math.round(val * 1024 * 1024);
  if (unit.startsWith("k")) return Math.round(val * 1024);
  return 500 * 1024 * 1024;
}

export function parseSpeedStringToMbps(speedStr: string | undefined): number {
  if (!speedStr) return 0;
  const cleaned = speedStr.trim().toLowerCase();
  if (
    cleaned.includes("finish") ||
    cleaned.includes("idle") ||
    cleaned.includes("done") ||
    cleaned.includes("offline") ||
    cleaned.includes("calculating")
  ) {
    return 0;
  }

  const numericMatch = cleaned.match(/([\d.]+)/);
  if (!numericMatch) return 0;

  const rawVal = parseFloat(numericMatch[1]);
  if (isNaN(rawVal)) return 0;

  if (cleaned.includes("gb/s") || cleaned.includes("gib/s") || cleaned.includes("gbyte/s")) {
    return rawVal * 1024;
  }
  if (cleaned.includes("mb/s") || cleaned.includes("mib/s") || cleaned.includes("mbyte/s")) {
    return rawVal;
  }
  if (cleaned.includes("kb/s") || cleaned.includes("kib/s") || cleaned.includes("kbyte/s")) {
    return rawVal / 1024;
  }
  if (cleaned.includes("b/s") || cleaned.includes("byte/s")) {
    return rawVal / (1024 * 1024);
  }
  return rawVal;
}

export async function triggerCodespaceWorkflow(): Promise<{ success: boolean; message: string }> {
  const token =
    process.env.GITHUB_DISPATCH_TOKEN ||
    process.env.GH_TOKEN ||
    process.env.GITHUB_TOKEN;
  const owner = process.env.GITHUB_REPO_OWNER || "tej-mega-third";
  const repo = process.env.GITHUB_REPO_NAME || "cloud-extracter";

  if (!token) {
    return {
      success: false,
      message: "Server configuration missing: GITHUB_DISPATCH_TOKEN is not defined in .env.local",
    };
  }

  try {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 8000);

    const response = await fetch(
      `https://api.github.com/repos/${owner}/${repo}/dispatches`,
      {
        method: "POST",
        headers: {
          Authorization: `Bearer ${token}`,
          Accept: "application/vnd.github+json",
          "X-GitHub-Api-Version": "2022-11-28",
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          event_type: "trigger_pipeline",
        }),
        signal: controller.signal,
      }
    );

    clearTimeout(timeoutId);

    // GitHub returns 204 No Content on a successful dispatch
    if (response.status === 204 || response.ok) {
      return {
        success: true,
        message: "Codespace wake signal dispatched successfully to GitHub Actions.",
      };
    }

    const errData = await response.json().catch(() => ({}));
    return {
      success: false,
      message: errData.message || `GitHub dispatch failed with HTTP ${response.status}`,
    };
  } catch (error: any) {
    return {
      success: false,
      message: error.name === "AbortError" ? "GitHub API request timed out" : error.message || "Network anomaly reaching GitHub API",
    };
  }
}

  const k = 1024;
  const dm = decimals < 0 ? 0 : decimals;

/**
 * Formats seconds into a human-readable ETA string (e.g., "42m", "12m 42s")
 */
export function formatEta(seconds: number): string {
  if (!seconds || seconds <= 0 || !isFinite(seconds)) return "-";
  const s = Math.floor(seconds);
  const minutes = Math.floor(s / 60);
  const hours = Math.floor(minutes / 60);

  if (hours > 0) {
    const remainingMins = minutes % 60;
    return `${hours}h ${remainingMins}m`;
  }
  if (minutes > 0) {
    const remainingSecs = s % 60;
    return remainingSecs > 0 ? `${minutes}m ${remainingSecs}s` : `${minutes}m`;
  }
  return `${s}s`;
}
